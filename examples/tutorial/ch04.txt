// examples/tutorial/ch04.txt
// Deeper -- Chapter 4: Dig!
//
// The core verb. Press B facing a diggable tile and the miner swings his pick; after a moment the tile
// turns to AIR -- and because the world grid is BOTH the picture and the collision map, the hole
// appears AND becomes walkable the instant we write it. One byte, two payoffs.
include "../../lib/io.lib";
include "../../lib/ppu.lib";
include "../../lib/scene.lib";
include "../../lib/anim.lib";
include "../../lib/platform.lib";
include "deeper_assets.gen.txt";
{
    const MAP_W = 24;   const MAP_H = 40;
    const T_AIR = 0;   const T_DIRT = 1;   const T_STONE = 2;

    const PAT_IDLE = 8;    const PAT_WALK = 10;   const PAT_JUMP = 13;   const PAT_MINE = 15;

    const F_DIG    = 4;    // tile_flags bit 2 = diggable
    const DIG_TIME = 30;   // frames of swinging before a tile breaks (~0.5s at 60fps)

    var byte world[960];
    var byte tile_flags[8] = { 0, 5, 1, 0, 0, 0, 0, 0 };   // air=0  dirt=solid+dig(5)  stone=solid(1)

    // Dig state: which tile we're currently working, and how many frames until it breaks.
    var int digging = 0;   var int dig_tx = 0;   var int dig_ty = 0;   var int dig_timer = 0;

    void build_world() {
        var int x; var int y; var int idx;
        y = 0;
        while y < MAP_H {
            x = 0;
            while x < MAP_W {
                idx = y * MAP_W + x;
                if x == 0              { world[idx] = T_STONE; }
                else if x == MAP_W - 1 { world[idx] = T_STONE; }
                else if y == 0         { world[idx] = T_STONE; }
                else if y == MAP_H - 1 { world[idx] = T_STONE; }
                else                   { world[idx] = T_DIRT; }
                x = x + 1;
            }
            y = y + 1;
        }
        y = 1;
        while y <= 3 { x = 1; while x < MAP_W - 1 { world[y * MAP_W + x] = T_AIR; x = x + 1; } y = y + 1; }
    }

    int main() {
        var int ix; var int jmp; var int pattr;
        var int tx; var int ty; var int t; var int dig_now;

        sys_ppu_dma(PAL_MASTER, PPU_PAL);
        ppu_upload_font(&font8x8);
        sys_ppu_dma(ANIM_DIRT_ID,       PPU_PAT + T_DIRT * 256);
        sys_ppu_dma(ANIM_STONE_ID,      PPU_PAT + T_STONE * 256);
        sys_ppu_dma(ANIM_MINER_ID,      PPU_PAT + PAT_IDLE * 256);
        sys_ppu_dma(ANIM_MINER_WALK_ID, PPU_PAT + PAT_WALK * 256);
        sys_ppu_dma(ANIM_MINER_JUMP_ID, PPU_PAT + PAT_JUMP * 256);
        sys_ppu_dma(ANIM_MINER_MINE_ID, PPU_PAT + PAT_MINE * 256);

        build_world();
        plat_bind(&world, MAP_W, MAP_H, &tile_flags);
        plat_spawn(5 * 16, 32);
        anim_play(0, PAT_IDLE, 2, clip_period[ANIM_MINER_ID]);
        sys_set_fps(60);

        while (true) {
            // --- input -> one physics step ---
            ix = 0;
            if button(BTN_LEFT)  { ix = 0 - 1; }
            if button(BTN_RIGHT) { ix = 1; }
            jmp = 0;
            if button_pressed(BTN_A) { jmp = 1; }
            plat_update(ix, jmp);

            // --- pick the target tile: the one just past the body in the faced direction, or (DOWN)
            //     the tile under the feet ---
            if button(BTN_DOWN) {
                tx = (plat_x + 8) >> 4;   ty = ((plat_y + 15) >> 4) + 1;   // under the feet
            } else { if button(BTN_UP) {
                tx = (plat_x + 8) >> 4;   ty = ((plat_y + 1) >> 4) - 1;     // above the head
            } else {
                if plat_facing == 1 { tx = (plat_x + 12) >> 4; } else { tx = (plat_x + 3) >> 4; }
                ty = (plat_y + 8) >> 4;                                     // in front, mid-body
            } }
            // is that tile in bounds AND diggable?
            dig_now = 0;
            if tx >= 0 { if tx < MAP_W { if ty >= 0 { if ty < MAP_H {
                t = world[ty * MAP_W + tx];
                if (tile_flags[t] & F_DIG) != 0 { if button(BTN_B) { dig_now = 1; } }
            } } } }

            // --- dig: swing while B is held on a diggable tile; break it when the timer runs out ---
            if dig_now == 1 {
                if dig_tx != tx { digging = 0; }              // aimed at a new tile -> start over
                if dig_ty != ty { digging = 0; }
                if digging == 0 { digging = 1; dig_tx = tx; dig_ty = ty; dig_timer = DIG_TIME; }
                dig_timer = dig_timer - 1;
                if dig_timer <= 0 {
                    world[dig_ty * MAP_W + dig_tx] = T_AIR;   // BREAK IT: picture + collision, one write
                    digging = 0;
                    // (a "thunk" sound effect goes here in Chapter 10)
                }
            } else {
                digging = 0;
            }

            // --- animation: digging beats moving; then jump / walk / idle ---
            if digging == 1 { anim_play(0, PAT_MINE, 5, clip_period[ANIM_MINER_MINE_ID]); }
            else { if plat_grounded == 0 { anim_play(0, PAT_JUMP, 2, clip_period[ANIM_MINER_JUMP_ID]); }
                   else { if ix == 0 { anim_play(0, PAT_IDLE, 2, clip_period[ANIM_MINER_ID]); }
                          else { anim_play(0, PAT_WALK, 3, clip_period[ANIM_MINER_WALK_ID]); } } }

            // --- draw ---
            cmd_reset();
            ppu_backdrop(0x0000);
            scene_stream(&world, MAP_W, MAP_H);
            scene_clear();
            pattr = 0;
            if plat_facing == (0 - 1) { pattr = 1; }
            scene_obj(anim_pat(0), plat_x, plat_y, pattr);
            scene_draw();

            buttons_update();
            ppu_present();
        }
        return 0;
    }
}
