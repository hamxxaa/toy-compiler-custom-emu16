// examples/tutorial/ch11.txt
// Deeper -- Chapter 11: save & load.
//
// The world remembers. Climb out of the mine and your tunnels, ore, money, health, and torch are written to
// a save slot -- RLE-compressed, because a 100x100 grid of mostly-identical tiles squashes tiny. Boot again
// and it all loads back: the mine is exactly as you left it.
include "../../lib/io.lib";
include "../../lib/ppu.lib";
include "../../lib/scene.lib";
include "../../lib/anim.lib";
include "../../lib/platform.lib";
include "../../lib/apu.lib";
include "../../lib/music.lib";
include "deeper_assets.gen.txt";
include "deeper_map.gen.txt";        // the mine: world[] / MAP_W / MAP_H / tile_flags[] (ore + lava baked in)
include "deeper_surface.gen.txt";    // the surface: SURF_world[] / SURF_MAP_W / SURF_MAP_H
{
    const T_AIR = 0;   const T_DIRT = 1;   const T_STONE = 2;   const T_DIRT_BG = 3;
    const T_GRASS = 4; const T_BASE = 5;   const T_LADDER = 6;  const T_ORE = 7;   const T_LAVA = 8;

    // Lava owns tile slot 8, so the miner's sprite clips live at slots 9 and up.
    const PAT_IDLE = 9;    const PAT_WALK = 11;   const PAT_JUMP = 14;   const PAT_MINE = 16;

    const F_DIG    = 4;      // tile_flags bit 2 = diggable
    const F_ORE    = 16;     // tile_flags bit 4 = gives ore
    const F_HAZARD = 8;      // tile_flags bit 3 = hurts on contact
    const DIG_TIME = 30;
    const SURFACE_ROW = 5;   // first dirt row in the mine -- DEPTH counts down from here
    const C_WHITE  = 21;

    const ORE_PRICE  = 5;
    const TORCH_COST = 15;
    const TORCH_STEP = 6;

    const MAX_HP    = 3;
    const INV_TIME  = 60;    // invincibility frames after a hit
    const FLASH_T   = 4;     // frames the miner flashes white on a hit
    const FALL_MIN  = 96;    // land MORE than 6 tiles below where you last stood -> hurt (a jump lands even)

    var int cur_world = 0;   var int cur_w = 0;   var int cur_h = 0;   var int cur_map = 0;
    var int on_ladder = 0;

    var int digging = 0;   var int dig_tx = 0;   var int dig_ty = 0;   var int dig_timer = 0;   var int dig_tile = 0;
    var int ore_count = 0; var int money = 0;    var int torch_r2 = 8;
    var int hp = 3;        var int inv = 0;
    var int flash_timer = 0;  var int flash_restore = 0;                // damage flash (ppu_pal1 -> restore)
    var int ground_y = 0;  var int was_air = 0;                         // last standing height + airborne flag

    var byte s_depth[7] = { 68, 69, 80, 84, 72, 58, 0 };                              // "DEPTH:"
    var byte s_ore[5]   = { 79, 82, 69, 58, 0 };                                      // "ORE:"
    var byte s_cash[3]  = { 36, 58, 0 };                                              // "$:"
    var byte s_hp[4]    = { 72, 80, 58, 0 };                                          // "HP:"
    var byte s_shop[13] = { 88, 61, 83, 69, 76, 76, 32, 89, 61, 66, 85, 89, 0 };      // "X=SELL Y=BUY"

    // --- audio: the song is pak asset NUM_CLIPS (appended right after the sprite/palette assets) ---
    const SONG_ID = NUM_CLIPS;
    var byte song_buf[2048];

    const INST_SFX_DIG  = 8;    // a short noise thunk when a tile breaks
    const INST_SFX_ORE  = 9;    // a bright chime when ore drops
    const INST_SFX_HURT = 10;   // a harsh descending buzz on damage
    var byte sfx_dig_vol[3]   = { 12, 6, 0 };
    var byte sfx_ore_vol[5]   = { 13, 10, 6, 3, 0 };
    var byte sfx_ore_duty[1]  = { 2 };
    var byte sfx_hurt_vol[5]  = { 14, 10, 6, 3, 0 };
    var byte sfx_hurt_duty[1] = { 1 };
    var byte sfx_hurt_arp[4]  = { 0, 253, 249, 244 };   // 0, -3, -7, -12 (signed: 256 - n) -> a downward slide

    // --- save/load: RLE the mine grid + the scalars into slot 0 ---
    const SAVE_MAX = 12288;                     // buffer capacity -- RLE keeps a real save far smaller
    const SAVE_SLOT = 0;
    var byte save_buf[SAVE_MAX];
    var byte s_saved[6] = { 83, 65, 86, 69, 68, 0 };   // "SAVED"
    var int saved_msg = 0;

    // Light-aware push (Chapter 7), using the upgradeable torch_r2 (Chapter 8).
    void light_stream(int wrld, int mw, int mh, int ptx, int pty) {
        var int tox; var int toy; var int r; var int c; var int tx; var int ty;
        var int dx; var int dy; var int tile;
        tox = scene_cam_x >> 4;   toy = scene_cam_y >> 4;
        ppu_scroll(scene_cam_x & 511, scene_cam_y & 511);
        r = 0;
        while r < 9 {
            ty = toy + r;
            if ty < mh {
                c = 0;
                while c < 11 {
                    tx = tox + c;
                    if tx < mw {
                        dx = tx - ptx;   if dx < 0 { dx = 0 - dx; }
                        dy = ty - pty;   if dy < 0 { dy = 0 - dy; }
                        tile = 0;
                        if dx * dx + dy * dy <= torch_r2 { tile = ppu_rdbyte(wrld + ty * mw + tx); }
                        cu8(3); cu8(tx & 31); cu8(ty & 31); cu8(1); cu8(1);   cu8(tile);
                    }
                    c = c + 1;
                }
            }
            r = r + 1;
        }
    }

    // How many tiles from `start` share world[start]'s value (capped at 255 -- one RLE run).
    int rle_run_len(int start, int total, int v) {
        var int run;
        run = 1;
        while start + run < total {
            if world[start + run] != v { return run; }
            if run >= 255 { return run; }
            run = run + 1;
        }
        return run;
    }

    // Pack {ore, money, hp, torch} + the RLE'd mine into save_buf and write it to the slot.
    int save_game() {
        var int off; var int i; var int run; var int total;
        total = MAP_W * MAP_H;
        save_buf[0] = ore_count & 255;    save_buf[1] = ore_count >> 8;    // 16-bit counters, lo then hi
        save_buf[2] = money & 255;        save_buf[3] = money >> 8;
        save_buf[4] = hp;
        save_buf[5] = torch_r2 & 255;     save_buf[6] = torch_r2 >> 8;
        off = 7;
        i = 0;
        while i < total {
            if off > SAVE_MAX - 2 { return 0; }              // never overflow the buffer -> abort, keep old save
            run = rle_run_len(i, total, world[i]);
            save_buf[off] = run;         off = off + 1;      // (count, tile) pair
            save_buf[off] = world[i];    off = off + 1;
            i = i + run;
        }
        return sys_save(&save_buf, off, SAVE_SLOT);          // write exactly the bytes we filled
    }

    // Reverse of save_game: restore the scalars and expand the RLE back into world[].
    void load_game() {
        var int off; var int i; var int run; var int v; var int k; var int n;
        n = sys_load(&save_buf, SAVE_MAX, SAVE_SLOT);        // bytes read (0 = no save in this slot)
        if n == 0 { return; }
        ore_count = save_buf[0] + save_buf[1] * 256;
        money     = save_buf[2] + save_buf[3] * 256;
        hp        = save_buf[4];
        torch_r2  = save_buf[5] + save_buf[6] * 256;
        off = 7;
        i = 0;
        while off < n {
            run = save_buf[off];   off = off + 1;
            v   = save_buf[off];   off = off + 1;
            k = 0;
            while k < run { world[i] = v; i = i + 1; k = k + 1; }
        }
    }

    void go_to_surface() {
        cur_world = &SURF_world;   cur_w = SURF_MAP_W;   cur_h = SURF_MAP_H;   cur_map = 0;
        plat_bind(cur_world, cur_w, cur_h, &tile_flags);
        plat_spawn(10 * 16, 14 * 16);   on_ladder = 1;   inv = INV_TIME;   ground_y = plat_y;   // grace + fall baseline
    }
    void go_to_mine() {
        cur_world = &world;   cur_w = MAP_W;   cur_h = MAP_H;   cur_map = 1;
        plat_bind(cur_world, cur_w, cur_h, &tile_flags);
        plat_spawn(29 * 16, 1 * 16);   on_ladder = 1;   inv = INV_TIME;   ground_y = plat_y;    // arrive at the mine ladder (col 29)
    }

    int main() {
        var int ix; var int jmp; var int pattr;
        var int tx; var int ty; var int t; var int dig_now; var int depth;
        var int gx; var int gy; var int gt; var int by; var int bt;
        var int hurt; var int fall_depth;

        sys_ppu_dma(PAL_MASTER, PPU_PAL);
        ppu_upload_font(&font8x8);
        sys_ppu_dma(ANIM_DIRT_ID,       PPU_PAT + T_DIRT * 256);
        sys_ppu_dma(ANIM_STONE_ID,      PPU_PAT + T_STONE * 256);
        sys_ppu_dma(ANIM_DIRT_BG_ID,    PPU_PAT + T_DIRT_BG * 256);
        sys_ppu_dma(ANIM_GRASS_ID,      PPU_PAT + T_GRASS * 256);
        sys_ppu_dma(ANIM_BASE_ID,       PPU_PAT + T_BASE * 256);
        sys_ppu_dma(ANIM_LADDER_ID,     PPU_PAT + T_LADDER * 256);
        sys_ppu_dma(ANIM_ORE_ID,        PPU_PAT + T_ORE * 256);
        sys_ppu_dma(ANIM_LAVA_ID,       PPU_PAT + T_LAVA * 256);
        sys_ppu_dma(ANIM_MINER_ID,      PPU_PAT + PAT_IDLE * 256);
        sys_ppu_dma(ANIM_MINER_WALK_ID, PPU_PAT + PAT_WALK * 256);
        sys_ppu_dma(ANIM_MINER_JUMP_ID, PPU_PAT + PAT_JUMP * 256);
        sys_ppu_dma(ANIM_MINER_MINE_ID, PPU_PAT + PAT_MINE * 256);

        // --- audio: register the three SFX instruments, then load + start the song ---
        apu_inst_vol(INST_SFX_DIG,  255, &sfx_dig_vol,   3);
        apu_inst_noise(INST_SFX_DIG, 0);
        apu_inst_vol(INST_SFX_ORE,  255, &sfx_ore_vol,   5);
        apu_inst_duty(INST_SFX_ORE, 255, &sfx_ore_duty,  1);
        apu_inst_vol(INST_SFX_HURT, 255, &sfx_hurt_vol,  5);
        apu_inst_duty(INST_SFX_HURT,255, &sfx_hurt_duty, 1);
        apu_inst_arp(INST_SFX_HURT, 255, &sfx_hurt_arp,  4);
        apu_master_vol(60);
        apu_submit();
        music_load_song(SONG_ID, &song_buf, 2048);   // loads the song, uploads its instruments, starts it

        if sys_save_exists(SAVE_SLOT) != 0 { load_game(); }   // resume: tunnels + ore + money + hp + torch
        else { hp = MAX_HP; }                                 // first run: the mine is the authored map as-is
        go_to_surface();
        anim_play(0, PAT_IDLE, 2, clip_period[ANIM_MINER_ID]);
        sys_set_fps(60);

        while (true) {
            ix = 0;
            if button(BTN_LEFT)  { ix = 0 - 1; }
            if button(BTN_RIGHT) { ix = 1; }
            jmp = 0;
            if button_pressed(BTN_A) { jmp = 1; }
            plat_update(ix, jmp);

            // --- ladder: drop into it on the surface to descend; press UP at it in the mine to climb out ---
            gx = (plat_x + 8) >> 4;   gy = (plat_y + 16) >> 4;   // tile under his feet
            by = (plat_y + 4) >> 4;                              // tile his upper body is in
            gt = 0;   bt = 0;
            if gx >= 0 { if gx < cur_w {
                if gy >= 0 { if gy < cur_h { gt = ppu_rdbyte(cur_world + gy * cur_w + gx); } }
                if by >= 0 { if by < cur_h { bt = ppu_rdbyte(cur_world + by * cur_w + gx); } }
            } }
            if cur_map == 0 {
                if gt == T_LADDER { if on_ladder == 0 { go_to_mine(); } on_ladder = 1; }   // fell into the hole
                else { if plat_grounded == 1 { on_ladder = 0; } }
            } else {
                if bt == T_LADDER { if button(BTN_UP) { go_to_surface();   save_game();   saved_msg = 40; } }
            }

            // --- fall distance: how far below the last spot he stood did he land this frame? ---
            if plat_grounded == 1 {
                if was_air == 1 { fall_depth = plat_y - ground_y; } else { fall_depth = 0; }
                ground_y = plat_y;   was_air = 0;
            } else { fall_depth = 0;   was_air = 1; }

            // --- damage: lava underfoot + a long drop, gated by i-frames; 0 HP -> wake at the surface ---
            if inv > 0 { inv = inv - 1; }
            else {
                hurt = 0;
                if plat_flag_at(plat_x + 8, plat_y + 16, F_HAZARD) != 0 { hurt = 1; }   // standing on lava
                if fall_depth > FALL_MIN { hurt = 1; }                                  // dropped a long way
                if hurt == 1 { hp = hp - 1; inv = INV_TIME; flash_timer = FLASH_T; sfx_play(1, N_A4, INST_SFX_HURT, 12, 3); }
            }
            if hp <= 0 { go_to_surface(); hp = MAX_HP; save_game(); saved_msg = 40; }   // waking up banks it too

            // --- shop: on the surface, by the base (cols 3..7) -- sell ore / buy torch ---
            if cur_map == 0 { if plat_x >= 3 * 16 { if plat_x <= 7 * 16 {
                if button_pressed(BTN_X) { money = money + ore_count * ORE_PRICE; ore_count = 0; }
                if button_pressed(BTN_Y) { if money >= TORCH_COST { money = money - TORCH_COST; torch_r2 = torch_r2 + TORCH_STEP; } }
            } } }

            // aim + dig (lava has no F_DIG, so it can't be dug -- you route around it)
            if button(BTN_DOWN) { tx = (plat_x + 8) >> 4; ty = ((plat_y + 15) >> 4) + 1; }
            else { if button(BTN_UP) { tx = (plat_x + 8) >> 4; ty = ((plat_y + 1) >> 4) - 1; }
                   else { if plat_facing == 1 { tx = (plat_x + 12) >> 4; } else { tx = (plat_x + 3) >> 4; }
                          ty = (plat_y + 8) >> 4; } }
            dig_now = 0;
            if tx >= 0 { if tx < cur_w { if ty >= 0 { if ty < cur_h {
                t = ppu_rdbyte(cur_world + ty * cur_w + tx);
                if (tile_flags[t] & F_DIG) != 0 { if button(BTN_B) { dig_now = 1; } }
            } } } }
            if dig_now == 1 {
                if dig_tx != tx { digging = 0; }
                if dig_ty != ty { digging = 0; }
                if digging == 0 { digging = 1; dig_tx = tx; dig_ty = ty; dig_timer = DIG_TIME; dig_tile = t; }
                dig_timer = dig_timer - 1;
                if dig_timer <= 0 {
                    if (tile_flags[dig_tile] & F_ORE) != 0 {
                        ore_count = ore_count + 1;
                        sfx_play(1, N_C6, INST_SFX_ORE, 10, 2);   // ore chime on a pulse channel
                    }
                    poke_byte(cur_world + dig_ty * cur_w + dig_tx, T_DIRT_BG);
                    sfx_play(3, N_C3, INST_SFX_DIG, 5, 1);         // dig thunk on the noise channel
                    digging = 0;
                }
            } else { digging = 0; }

            if digging == 1 { anim_play(0, PAT_MINE, 5, clip_period[ANIM_MINER_MINE_ID]); }
            else { if plat_grounded == 0 { anim_play(0, PAT_JUMP, 2, clip_period[ANIM_MINER_JUMP_ID]); }
                   else { if ix == 0 { anim_play(0, PAT_IDLE, 2, clip_period[ANIM_MINER_ID]); }
                          else { anim_play(0, PAT_WALK, 3, clip_period[ANIM_MINER_WALK_ID]); } } }

            scene_follow(plat_x, plat_y, cur_w, cur_h);

            cmd_reset();
            ppu_backdrop(0x0000);
            if cur_map == 1 { light_stream(cur_world, cur_w, cur_h, (plat_x + 8) >> 4, (plat_y + 8) >> 4); }
            else            { scene_stream(cur_world, cur_w, cur_h); }
            scene_clear();
            pattr = 0;
            if plat_facing == (0 - 1) { pattr = 1; }
            scene_obj(anim_pat(0), plat_x, plat_y, pattr);
            scene_draw();

            // --- damage flash: whiten the miner's palette entries for a few frames, then restore (PRAM) ---
            if flash_timer > 0 {
                ppu_pal1(3, 0xFFFF);  ppu_pal1(4, 0xFFFF);  ppu_pal1(6, 0xFFFF);
                ppu_pal1(8, 0xFFFF);  ppu_pal1(18, 0xFFFF);
                flash_timer = flash_timer - 1;
                if flash_timer == 0 { flash_restore = 1; }
            } else { if flash_restore == 1 { sys_ppu_dma(PAL_MASTER, PPU_PAL); flash_restore = 0; } }

            // --- HUD: ore + money + HP always; DEPTH underground; shop hint at the base ---
            tile_clear(0, 0, 20, 2);             // wipe last frame's HUD so stale DEPTH/shop text can't linger
            tile_text(0, 1, &s_ore, C_WHITE);    tile_number(4, 1, ore_count, C_WHITE);
            tile_text(9, 1, &s_cash, C_WHITE);   tile_number(11, 1, money, C_WHITE);
            tile_text(15, 1, &s_hp, C_WHITE);    tile_number(18, 1, hp, C_WHITE);
            if cur_map == 1 {
                depth = ((plat_y + 15) >> 4) - SURFACE_ROW;
                if depth < 0 { depth = 0; }
                tile_text(0, 0, &s_depth, C_WHITE);   tile_number(7, 0, depth, C_WHITE);
            } else { if plat_x >= 3 * 16 { if plat_x <= 7 * 16 { tile_text(0, 0, &s_shop, C_WHITE); } } }
            if saved_msg > 0 { tile_text(7, 0, &s_saved, C_WHITE); saved_msg = saved_msg - 1; }

            music_frame();      // advance the sequencer + service any live SFX (after all sfx_play calls)
            buttons_update();
            ppu_present();
        }
        return 0;
    }
}
