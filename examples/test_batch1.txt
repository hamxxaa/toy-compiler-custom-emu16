{
    int conditional(int x) {
        var int result = 0;
        if x > 10 do {
            result = 1;
        }
        if x < 5 do {
            result = 2;
        }
        if x == 7 do {
            result = 3;
        }
        return result;
    }

    int loop_sum(int n) {
        var int i = 0;
        var int total = 0;
        while i < n do {
            total = total + i;
            i = i + 1;
        }
        return total;
    }

    int countdown(int n) {
        var int result = 0;
        while n > 0 do {
            result = result + n;
            n = n - 1;
        }
        return result;
    }

    int fibonacci(int n) {
        var int a = 0;
        var int b = 1;
        var int i = 0;
        while i < n do {
            var int temp = b;
            b = a + b;
            a = temp;
            i = i + 1;
        }
        return a;
    }

    int main() {
        var int pass = 0;
        var int fail = 0;

        var int t1 = conditional(15);
        if t1 == 1 do { pass = pass + 1; }
        if t1 != 1 do { fail = fail + 1; }

        var int t2 = conditional(3);
        if t2 == 2 do { pass = pass + 1; }
        if t2 != 2 do { fail = fail + 1; }

        var int t3 = loop_sum(5);
        if t3 == 10 do { pass = pass + 1; }
        if t3 != 10 do { fail = fail + 1; }

        var int t4 = countdown(5);
        if t4 == 15 do { pass = pass + 1; }
        if t4 != 15 do { fail = fail + 1; }

        var int t5 = fibonacci(10);
        if t5 == 55 do { pass = pass + 1; }
        if t5 != 55 do { fail = fail + 1; }

        var int t6 = fibonacci(0);
        if t6 == 0 do { pass = pass + 1; }
        if t6 != 0 do { fail = fail + 1; }

        if fail == 0 do {
            return pass;
        }
        return 0 - fail;
    }
}
