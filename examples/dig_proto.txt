    const PAT_PLAYER = 8;

    const T_AIR  = 0;   const T_DIRT = 1;   const T_ROCK = 2;
    const F_DIG  = 4;                          // tile-flags bit: diggable

    // ── Baked palette (RGB565, little-endian) ────────────────────────────────────────────────────
    var byte game_pal[16] = {

    };

    };
    var byte pat_rock[256] = {

    // tile id -> properties (bit0 solid, bit2 diggable). air=0, dirt=solid+dig, rock=solid.
    var byte tile_flags[8] = { 0, 5, 1, 0, 0, 0, 0, 0 };

    var byte world[960];   // MAP_W * MAP_H, filled by build_world()

    // Solid bedrock border ring, dirt fill, and a carved starting pocket (rows 1..5) to stand in.
    void build_world() {
        var int x; var int y; var int idx;
        y = 0;
        while y < MAP_H {
            x = 0;
            while x < MAP_W {
                idx = y * MAP_W + x;
                if x == 0            { world[idx] = T_ROCK; }
                else if x == MAP_W - 1 { world[idx] = T_ROCK; }
                else if y == 0         { world[idx] = T_ROCK; }
                else if y == MAP_H - 1 { world[idx] = T_ROCK; }
                else                   { world[idx] = T_DIRT; }
                x = x + 1;
            }
            y = y + 1;
        }
        y = 1;
        while y <= 5 {
            x = 1;
            while x < MAP_W - 1 { world[y * MAP_W + x] = T_AIR; x = x + 1; }
            y = y + 1;
        }
    }

    // Dig the tile in the held direction (default: straight down below the feet). Diggable -> air.
    void do_dig() {
        var int tx; var int ty; var int idx; var int cx; var int t;
        cx = (plat_x + 8) >> 4;                         // player's center column
        tx = cx;  ty = ((plat_y + 14) >> 4) + 1;        // default: tile below the feet
        if button(BTN_UP)    { tx = cx; ty = ((plat_y + 2) >> 4) - 1; }
        if button(BTN_LEFT)  { tx = ((plat_x + 4) >> 4) - 1;  ty = (plat_y + 8) >> 4; }
        if button(BTN_RIGHT) { tx = ((plat_x + 11) >> 4) + 1; ty = (plat_y + 8) >> 4; }
        if tx < 0 { return; }
        if tx >= MAP_W { return; }
        if ty < 0 { return; }
        if ty >= MAP_H { return; }
        idx = ty * MAP_W + tx;
        t = world[idx];
        if (tile_flags[t] & F_DIG) != 0 { world[idx] = T_AIR; }
    }

    int main() {
        var int ix; var int jmp; var int pattr;

        sys_ppu_upload(PPU_PAL, &game_pal, 16);
        ppu_upload_font(&font8x8);
        sys_ppu_upload(PPU_PAT + T_DIRT * 256,      &pat_dirt,   256);
        sys_ppu_upload(PPU_PAT + T_ROCK * 256,      &pat_rock,   256);
        sys_ppu_upload(PPU_PAT + PAT_PLAYER * 256,  &pat_player, 256);

        build_world();
        plat_bind(&world, MAP_W, MAP_H, &tile_flags);
        plat_spawn(12 * 16, 2 * 16);
        sys_set_fps(60);

        while (true) {
            ix = 0;
            if button(BTN_LEFT)  { ix = 0 - 1; }
            if button(BTN_RIGHT) { ix = 1; }
            jmp = 0;
            if button_pressed(BTN_A) { jmp = 1; }
            plat_update(ix, jmp);
            if button_pressed(BTN_B) { do_dig(); }

            scene_follow(plat_x, plat_y, MAP_W, MAP_H);
            cmd_reset();
            ppu_backdrop(0x0000);
            scene_stream(&world, MAP_W, MAP_H);
            scene_clear();
            pattr = 0;
            if plat_facing == (0 - 1) { pattr = 1; }
            scene_obj(PAT_PLAYER, plat_x, plat_y, pattr);
            scene_draw();

            tile_number(0, 0, plat_y >> 4, 1);      // tile row (depth proxy)
            tile_number(0, 1, plat_grounded, 1);    // grounded flag

            buttons_update();
            ppu_present();
        }
        return 0;
    }
