include "../lib/io.lib";

{
    int main() {
        
        while (true){
        var byte input = read_input();
        if input & 1 == 1 {
            fill(1);
        } else if input & 2 == 2 {
            fill(2);
        } else if input & 4 == 4 {
            fill(3);
        } else if input & 8 == 8 {
            fill(4);
        } else if input & 16 == 16 {
            fill(5);
        } else if input & 32 == 32 {
            fill(6);
        } else if input & 64 == 64 {
            fill(7);
        } else if input & 128 == 128 {
            fill(8);
        } else {
            fill(0);
        }
        }
        return 0;
    }
}
