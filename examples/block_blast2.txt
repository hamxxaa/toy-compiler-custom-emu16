include "../lib/io.lib";
include "../lib/sys.lib";
include "../lib/game.lib";

{
    var byte bricks[99];

    var byte numbers[50] = {
    0xF8, 0x88, 0x88, 0x88, 0xF8,
    0x08, 0x08, 0x08, 0x08, 0x08,
    0xF8, 0x08, 0xF8, 0x80, 0xF8,
    0xF8, 0x08, 0xF8, 0x08, 0xF8,
    0x10, 0x30, 0x50, 0xF8, 0x10,
    0xF8, 0x80, 0xF8, 0x08, 0xF8,
    0xF8, 0x80, 0xF8, 0x88, 0xF8,
    0x78, 0x08, 0x08, 0x08, 0x08,
    0xF8, 0x88, 0xF8, 0x88, 0xF8,
    0xF8, 0x88, 0xF8, 0x08, 0xF8 };

    var int map_width      = 11;
    var int map_height     = 8;
    var int brick_y_offset = 10;

    var int ball_shootingx = 82;
    var int ball_shootingy = 125;

    var int gameState = 0;  // 0=aim, 1=flight, 2=game over

    var byte ball_sprite[2] = { 0xC0, 0xC0 };

    var byte block_sprite[28] = {
    0xFF, 0xFC,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0x80, 0x04,
    0xFF, 0xFC };

    // Solid 14x14 sprite used to black-erase a brick cell
    var byte erase_sprite[28] = {
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC,
    0xFF, 0xFC };

    // Aim directions: fixed-point x128, 2 px/sub-step magnitude
    var int aimx[31] = { -247,-241,-232,-222,-210,-196,-181,-165,-147,-128,-108,-88,-66,-44,-22,0,22,44,66,88,108,128,147,165,181,196,210,222,232,241,247 };
    var int aimy[31] = { -66,-88,-108,-128,-147,-165,-181,-196,-210,-222,-232,-241,-247,-252,-255,-256,-255,-252,-247,-241,-232,-222,-210,-196,-181,-165,-147,-128,-108,-88,-66 };

    // Ball state – up to 50 balls; positions/velocities are fixed-point x128
    var int  ball_px[50];
    var int  ball_py[50];
    var int  ball_vx[50];
    var int  ball_vy[50];
    var byte ball_on[50];
    var int  ball_dx[50];   // last pixel drawn (x)
    var int  ball_dy[50];   // last pixel drawn (y)

    // Volley state
    var int aim          = 15;
    var int shot_vx      = 0;
    var int shot_vy      = 0;
    var int balls_left   = 0;
    var int launch_timer = 0;

    // Progression
    var int level        = 0;
    var int destroyed    = 0;
    var int ball_count   = 1;
    var int next_ball_at = 10;
    var int landed       = 0;   // has the first ball returned?
    var int next_x       = 82;  // x of first returned ball

    // Physics
    var int speed = 3;   // physics sub-steps per update_balls call
    var int ff    = 1;   // fast-forward multiplier (1 or 3)

    // Game-over screen strings (ASCII)
    var byte s_gameover[9] = { 71, 65, 77, 69, 32, 79, 86, 69, 82 };  // "GAME OVER"
    var byte s_pressa[7]   = { 80, 82, 69, 83, 83, 32, 65 };           // "PRESS A"

    // ---------------------------------------------------------------
    // Helpers
    // ---------------------------------------------------------------

    int cell_brick(int px, int py) {
        var int col = px / 14;
        var int row;
        if col < 0          { return 0; }
        if col >= map_width { return 0; }
        if py < brick_y_offset { return 0; }
        row = (py - brick_y_offset) / 14;
        if row >= map_height { return 0; }
        return bricks[row * map_width + col];
    }

    int active_count() {
        var int b = 0;
        var int c = 0;
        while b < 50 {
            if ball_on[b] == 1 { c = c + 1; }
            b = b + 1;
        }
        return c;
    }

    // Draw one brick cell (uses bricks[] for current HP)
    int draw_brick(int col, int row) {
        var int x     = col * 14;
        var int y     = row * 14 + brick_y_offset;
        var int value = bricks[row * map_width + col];
        var int tens  = value / 10;
        var int ones  = value - (tens * 10);
        draw_sprite(&block_sprite, 14, 14, x, y, 1);
        draw_sprite(&numbers + (tens * 5), 5, 5, x + 1, y + 4, 1);
        draw_sprite(&numbers + (ones * 5), 5, 5, x + 7, y + 4, 1);
        return 0;
    }

    // Erase one brick cell to black
    int erase_brick(int col, int row) {
        draw_sprite(&erase_sprite, 14, 14, col * 14, row * 14 + brick_y_offset, 0);
        return 0;
    }

    int draw_bricks() {
        var int i = 0;
        var int j;
        while i < map_height {
            j = 0;
            while j < map_width {
                if bricks[i * map_width + j] != 0 {
                    draw_brick(j, i);
                }
                j = j + 1;
            }
            i = i + 1;
        }
        return 0;
    }

    // Draw n right-aligned; x is the pixel position of the ones digit
    int draw_number(int n, int x, int y, int color) {
        var int d;
        var int dx    = x;
        var int safety = 0;
        if n == 0 {
            draw_char(48, dx, y, color);
            return 0;
        }
        while n > 0 {
            d = n - (n / 10) * 10;
            n = n / 10;
            draw_char(48 + d, dx, y, color);
            dx = dx - 8;
            safety = safety + 1;
            if safety > 5 { n = 0; }
        }
        return 0;
    }

    // HUD: ball count top-left, score top-right (drawn at y=0 on top of bricks)
    int draw_hud() {
        draw_char(66, 0, 1, 1);            // 'B'
        draw_char(58, 8, 1, 1);            // ':'
        draw_number(ball_count, 36, 1, 1);
        draw_char(83, 88, 1, 1);           // 'S'
        draw_char(58, 96, 1, 1);           // ':'
        draw_number(destroyed, 152, 1, 1);
        return 0;
    }

    int redraw_scene() {
        fill(0);
        draw_bricks();
        draw_sprite(&ball_sprite, 2, 2, ball_shootingx, ball_shootingy, 2);
        draw_hud();
        return 0;
    }

    int draw_aim() {
        var int tx   = ball_shootingx << 7;
        var int ty   = ball_shootingy << 7;
        var int tvx  = aimx[aim];
        var int tvy  = aimy[aim];
        var int done = 0;
        var int count = 0;
        var int px;
        var int py;
        while done == 0 {
            tx = tx + tvx;
            ty = ty + tvy;
            if tx < 0     { tx = 0;     tvx = 0 - tvx; }
            if tx > 20352 { tx = 20352; tvx = 0 - tvx; }
            if ty < 0     { done = 1; }
            px = tx >> 7;
            py = ty >> 7;
            if cell_brick(px, py) != 0 { done = 1; }
            if (count & 7) == 0 { plot(px, py, 1); }
            count = count + 1;
            if count > 300 { done = 1; }
        }
        return 0;
    }

    int draw_gameover() {
        fill(0);
        draw_string(&s_gameover, 9, 44, 44, 1);
        draw_number(destroyed, 120, 64, 1);
        draw_string(&s_pressa, 7, 52, 84, 1);
        return 0;
    }

    // ---------------------------------------------------------------
    // Progression
    // ---------------------------------------------------------------

    // Fill row 0 randomly; guarantee >=1 gap and >=1 brick
    int spawn_row() {
        var int c;
        var int base_hp = 1 + level / 3;
        var int hp;
        var int all_full;
        var int all_empty;
        var int fc;
        c = 0;
        while c < map_width {
            if rand_range(9) < 5 {
                hp = base_hp + rand_range(2);
                bricks[c] = hp;
            } else {
                bricks[c] = 0;
            }
            c = c + 1;
        }
        // Guarantee at least one gap
        all_full = 1;
        c = 0;
        while c < map_width {
            if bricks[c] == 0 { all_full = 0; }
            c = c + 1;
        }
        if all_full == 1 {
            fc = rand_range(map_width);
            bricks[fc] = 0;
        }
        // Guarantee at least one brick
        all_empty = 1;
        c = 0;
        while c < map_width {
            if bricks[c] != 0 { all_empty = 0; }
            c = c + 1;
        }
        if all_empty == 1 {
            fc = rand_range(map_width);
            bricks[fc] = base_hp;
        }
        return 0;
    }

    // Called at end of each volley: check game over, shift rows, spawn new top row
    int advance_level() {
        var int r;
        var int c;
        var int gameover = 0;
        // Game over if any brick is in the death row (row map_height-1)
        c = 0;
        while c < map_width {
            if bricks[(map_height - 1) * map_width + c] != 0 { gameover = 1; }
            c = c + 1;
        }
        if gameover == 1 {
            gameState = 2;
            return 0;
        }
        // Shift all rows down one (bottom to top so we don't clobber)
        r = map_height - 1;
        while r >= 1 {
            c = 0;
            while c < map_width {
                bricks[r * map_width + c] = bricks[(r - 1) * map_width + c];
                c = c + 1;
            }
            r = r - 1;
        }
        // Clear row 0 and spawn a new one
        c = 0;
        while c < map_width {
            bricks[c] = 0;
            c = c + 1;
        }
        spawn_row();
        // Ball rewards: one extra ball per 10 bricks destroyed
        while destroyed >= next_ball_at {
            ball_count   = ball_count + 1;
            next_ball_at = next_ball_at + 10;
        }
        if ball_count > 49 { ball_count = 49; }
        level = level + 1;
        return 0;
    }

    int reset_game() {
        var int i = 0;
        var int b = 0;
        while i < 99 {
            bricks[i] = 0;
            i = i + 1;
        }
        level          = 0;
        destroyed      = 0;
        ball_count     = 1;
        next_ball_at   = 10;
        ball_shootingx = 82;
        next_x         = 82;
        landed         = 0;
        aim            = 15;
        balls_left     = 0;
        launch_timer   = 0;
        while b < 50 {
            ball_on[b] = 0;
            b = b + 1;
        }
        spawn_row();
        gameState = 0;
        return 0;
    }

    // ---------------------------------------------------------------
    // Ball update: erase → sub-step physics → redraw
    // ---------------------------------------------------------------

    int update_balls() {
        var int b      = 0;
        var int s;
        var int steps  = speed * ff;
        var int opx;
        var int opy;
        var int nx;
        var int ny;
        var int col;
        var int row;
        var int oldcol;
        var int oldrow;
        var int idx;
        var int dx;
        var int dy;

        while b < 50 {
            if ball_on[b] == 1 {

                // 1. Erase ball at last drawn position (check each pixel to avoid
                //    blackening brick digit pixels when ball straddles a cell edge)
                dx = ball_dx[b];
                dy = ball_dy[b];
                if cell_brick(dx,     dy    ) != 0 { draw_brick(dx / 14,       (dy     - brick_y_offset) / 14); } else { plot(dx,     dy,     0); }
                if cell_brick(dx + 1, dy    ) != 0 { draw_brick((dx + 1) / 14, (dy     - brick_y_offset) / 14); } else { plot(dx + 1, dy,     0); }
                if cell_brick(dx,     dy + 1) != 0 { draw_brick(dx / 14,       (dy + 1 - brick_y_offset) / 14); } else { plot(dx,     dy + 1, 0); }
                if cell_brick(dx + 1, dy + 1) != 0 { draw_brick((dx + 1) / 14, (dy + 1 - brick_y_offset) / 14); } else { plot(dx + 1, dy + 1, 0); }

                // 2. Physics sub-steps (speed * ff iterations of 2 px each)
                s = 0;
                while s < steps {
                    opx = ball_px[b];
                    opy = ball_py[b];

                    ball_px[b] = ball_px[b] + ball_vx[b];
                    ball_py[b] = ball_py[b] + ball_vy[b];

                    if ball_px[b] < 384   { ball_px[b] = 384;   ball_vx[b] = 0 - ball_vx[b]; }
                    if ball_px[b] > 19840 { ball_px[b] = 19840; ball_vx[b] = 0 - ball_vx[b]; }
                    if ball_py[b] < 1280  { ball_py[b] = 1280;  ball_vy[b] = 0 - ball_vy[b]; }

                    nx = ball_px[b] >> 7;
                    ny = ball_py[b] >> 7;

                    if ny >= 127 {
                        ball_on[b] = 0;
                        if landed == 0 { next_x = nx; landed = 1; }
                        s = steps;  // exit sub-step loop
                    } else {
                                               col = nx / 14;
                        if (col >= 0) && (col < map_width) {
                            if ny >= brick_y_offset {
                                row = (ny - brick_y_offset) / 14;
                                if (row >= 0) && (row < map_height) {
                                    idx = row * map_width + col;
                                    if bricks[idx] != 0 {
                                        bricks[idx] = bricks[idx] - 1;
                                        erase_brick(col, row);
                                        if bricks[idx] == 0 {
                                            destroyed = destroyed + 1;
                                        } else {
                                            draw_brick(col, row);
                                        }
                                        oldcol = (opx >> 7) / 14;
                                        if (opy >> 7) < brick_y_offset {
                                            oldrow = 0 - 1;
                                        } else {
                                            oldrow = ((opy >> 7) - brick_y_offset) / 14;
                                        }
                                        if oldrow != row { ball_vy[b] = 0 - ball_vy[b]; }
                                        if oldcol != col { ball_vx[b] = 0 - ball_vx[b]; }
                                        ball_px[b] = opx;
                                        ball_py[b] = opy;
                                    }
                                }
                            }
                        }
                        s = s + 1;
                    }
                }

                // 3. Draw ball at new position (if still alive)
                if ball_on[b] == 1 {
                    nx = ball_px[b] >> 7;
                    ny = ball_py[b] >> 7;
                    draw_sprite(&ball_sprite, 2, 2, nx, ny, 3);
                    ball_dx[b] = nx;
                    ball_dy[b] = ny;
                }
            }
            b = b + 1;
        }
        return 0;
    }

    // ---------------------------------------------------------------
    // Main loop: 3-state machine
    // ---------------------------------------------------------------

    int main() {
        var int b = 0;
        var int slot;
        var int scene_dirty = 1;
        var int old_aim;

        srand_time();

        while b < 50 {
            ball_on[b] = 0;
            ball_dx[b] = ball_shootingx;
            ball_dy[b] = ball_shootingy;
            b = b + 1;
        }

        spawn_row();

        while (true) {
            if gameState == 0 {
                // ---- Aim ----
                old_aim = aim;
                if button_pressed(BTN_LEFT)  { if aim > 0  { aim = aim - 1; } }
                if button_pressed(BTN_RIGHT) { if aim < 30 { aim = aim + 1; } }

                if (aim != old_aim) || (scene_dirty == 1) {
                    redraw_scene();
                    draw_aim();
                    scene_dirty = 0;
                }

                if button_pressed(BTN_A) {
                    shot_vx      = aimx[aim];
                    shot_vy      = aimy[aim];
                    balls_left   = ball_count;
                    launch_timer = 0;
                    landed       = 0;
                    ff           = 1;
                    gameState    = 1;
                    redraw_scene();
                }

            } else if gameState == 1 {
                // ---- Flight ----

                // B: instant recall – end the volley now
                if button_pressed(BTN_B) {
                    b = 0;
                    while b < 50 {
                        ball_on[b] = 0;
                        b = b + 1;
                    }
                    balls_left = 0;
                }

                // X: fast-forward (3x sub-steps this frame)
                ff = 1;
                if button_pressed(BTN_X) { ff = 3; }

                // Staggered launch: one ball every 4 frames
                if balls_left > 0 {
                    if launch_timer == 0 {
                        slot = ball_count - balls_left;
                        ball_px[slot] = ball_shootingx << 7;
                        ball_py[slot] = ball_shootingy << 7;
                        ball_vx[slot] = shot_vx;
                        ball_vy[slot] = shot_vy;
                        ball_on[slot] = 1;
                        ball_dx[slot] = ball_shootingx;
                        ball_dy[slot] = ball_shootingy;
                        balls_left    = balls_left - 1;
                        launch_timer  = 4;
                    } else {
                        launch_timer = launch_timer - 1;
                    }
                }

                update_balls();

                // Check volley end
                if balls_left == 0 {
                    if active_count() == 0 {
                        // Reposition launcher to where first ball landed
                        if landed == 1 {
                            ball_shootingx = next_x;
                            if ball_shootingx < 2   { ball_shootingx = 2;   }
                            if ball_shootingx > 156 { ball_shootingx = 156; }
                        }
                        advance_level();
                        if gameState == 2 {
                            draw_gameover();
                        } else {
                            scene_dirty = 1;
                            gameState   = 0;
                        }
                    }
                }

            } else {
                // ---- Game Over ----
                if button_pressed(BTN_A) {
                    reset_game();
                    scene_dirty = 1;
                }
            }

            buttons_update();
            sys_present();
        }
        return 0;
    }
}
