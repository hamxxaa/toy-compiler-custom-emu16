// ─────────────────────────────────────────────────────────────────────────────────────────────
// PPU overworld — the Phase 3 reference scene, built entirely on the PPU model.
//
// The CPU never draws a pixel: each frame it fills a command buffer (scroll + OAM + text) and the
// PPU composites a scrolling tile world + Y-sorted sprites + a text HUD/dialog. Shows: walking a
// tile world, tile collision (water blocks), animated + flipped hero, walk-behind trees (Y-sort),
// a colored HUD, a dialog box (button B), and a live palette flash (button A).
//
// Assets stream from ppu_overworld.pak (built from assets/sprites/sprites.list; ids below match that order).
//
// Build + run:
//   python main.py examples/ppu_overworld.txt --save-rom ppu_overworld
//   python tools/image_import.py assets/sprites/sprites.list build/demo_assets
//   python tools/pack_assets.py build/demo_assets/assets.manifest build/roms/ppu_overworld
//   ./pc_emu.exe --rom build/roms/ppu_overworld.rom --frames 1     # headless (writes frame.ppm)
//   # browser: open simulator/index.html and drop ppu_overworld.rom + ppu_overworld.pak
// Controls: arrows move, B toggles a dialog box, A flashes the palette.
// ─────────────────────────────────────────────────────────────────────────────────────────────
include "../lib/io.lib";      // read_input
include "../lib/sys.lib";     // sys_time (animation clock)
include "../lib/ppu.lib";     // PPU command builder + text plane (also font8x8, for the text plane)
include "../lib/map.lib";     // map_solid_at (tile collision against the compiled-in world)

{
    // ---- asset ids (order == assets/sprites/sprites.list, packed into ppu_overworld.pak) ----
    const PAL_MASTER  = 0;
    const HERO_SIDE_A = 1;    // "Walk" (side, faces right), 4 frames
    const HERO_UP_A   = 3;    // "walk_up",   4 frames
    const HERO_DOWN_A = 4;    // "walk_down", 4 frames
    const TREE1_A     = 8;
    const TREE2_A     = 9;
    const TREE3_A     = 10;
    const TREE4_A     = 11;

    // ---- PPU pattern slots (16x16 patterns; sprites + bg tiles share the pattern table) ----
    const S_HERO_SIDE = 0;    // slots 0..3
    const S_HERO_UP   = 4;    // slots 4..7
    const S_HERO_DOWN = 8;    // slots 8..11
    const S_TREE1     = 12;
    const S_TREE2     = 13;
    const S_TREE3     = 14;
    const S_TREE4     = 15;
    const T_GRASS     = 16;
    const T_WATER     = 17;   // the solid tile
    const T_FLOWER    = 18;

    // ---- palette indices (master.gpl order) ----
    const C_GREEN  = 10;
    const C_DGREEN = 12;
    const C_BLUE   = 18;
    const C_YELLOW = 8;
    const C_WHITE  = 21;
    const C_BLACK  = 0;

    // ---- world: 32x32 tiles == the PPU tilemap window (512x512 px). Doubles as the CPU collision map. ----
    const MAPW = 32;
    const MAPH = 32;
    var byte world[1024];
    var byte tile_flags[32];
    var byte tilebuf[256];        // scratch for building a tile pattern before upload

    // ---- hero + camera ----
    var int hx; var int hy;       // world pixel pos (top-left of the 16x16 sprite)
    var int face;                 // 0 down, 1 up, 2 side
    var int flip;                 // 1 = facing left
    var int moving;

    // ---- Y-sort object list (index 0 = hero, 1..4 = trees) ----
    const NOBJ = 5;
    var int obj_pat[NOBJ];
    var int obj_x[NOBJ];
    var int obj_y[NOBJ];
    var int obj_attr[NOBJ];
    var int order[NOBJ];

    // ---- ui ----
    var int dialog_on; var int prev_b;
    var int flash_t;   var int flash_restore; var int prev_a;

    // Build a 256-byte tile pattern (solid `base`, sparse `speckle`) and upload it to a pattern slot.
    void make_tile(int slot, int base, int speckle) {
        var int i; var int k;
        i = 0;
        while i < 256 {
            k = i & 31;
            if k == 5 { tilebuf[i] = speckle; }
            else { if k == 22 { tilebuf[i] = speckle; } else { tilebuf[i] = base; } }
            i = i + 1;
        }
        sys_ppu_upload(PPU_PAT + slot * 256, &tilebuf, 256);
    }

    // Fill the 32x32 world: grass field, solid water border + a central lake, a few flower tiles.
    void build_world() {
        var int r; var int c; var int idx; var int t;
        r = 0;
        while r < MAPH {
            c = 0;
            while c < MAPW {
                idx = r * MAPW + c;
                t = T_GRASS;
                if r == 0 { t = T_WATER; }
                if r == MAPH - 1 { t = T_WATER; }
                if c == 0 { t = T_WATER; }
                if c == MAPW - 1 { t = T_WATER; }
                world[idx] = t;
                c = c + 1;
            }
            r = r + 1;
        }
        r = 6;
        while r < 11 {
            c = 8;
            while c < 15 {
                world[r * MAPW + c] = T_WATER;
                c = c + 1;
            }
            r = r + 1;
        }
        world[4 * MAPW + 20] = T_FLOWER;
        world[5 * MAPW + 21] = T_FLOWER;
        world[18 * MAPW + 6] = T_FLOWER;
    }

    void build_flags() {
        var int i;
        i = 0;
        while i < 32 { tile_flags[i] = 0; i = i + 1; }
        tile_flags[T_WATER] = 1;
    }

    // Lower-body footprint corner test -> 1 if any corner is on a solid tile.
    int blocked(int px, int py) {
        var int x0; var int x1; var int y0; var int y1;
        x0 = px + 3;  x1 = px + 12;
        y0 = py + 10; y1 = py + 15;
        if map_solid_at(&world, MAPW, &tile_flags, x0, y0) == 1 { return 1; }
        if map_solid_at(&world, MAPW, &tile_flags, x1, y0) == 1 { return 1; }
        if map_solid_at(&world, MAPW, &tile_flags, x0, y1) == 1 { return 1; }
        if map_solid_at(&world, MAPW, &tile_flags, x1, y1) == 1 { return 1; }
        return 0;
    }

    int main() {
        var int input; var int a_now; var int b_now;
        var int dx; var int dy; var int nx; var int ny;
        var int now; var int frame; var int base;
        var int camx; var int camy;
        var int i; var int j; var int t; var int oi;

        // ---- stream assets into the PPU ----
        sys_ppu_dma(PAL_MASTER, PPU_PAL);
        sys_ppu_dma(HERO_SIDE_A, PPU_PAT + S_HERO_SIDE * 256);
        sys_ppu_dma(HERO_UP_A,   PPU_PAT + S_HERO_UP * 256);
        sys_ppu_dma(HERO_DOWN_A, PPU_PAT + S_HERO_DOWN * 256);
        sys_ppu_dma(TREE1_A, PPU_PAT + S_TREE1 * 256);
        sys_ppu_dma(TREE2_A, PPU_PAT + S_TREE2 * 256);
        sys_ppu_dma(TREE3_A, PPU_PAT + S_TREE3 * 256);
        sys_ppu_dma(TREE4_A, PPU_PAT + S_TREE4 * 256);
        ppu_upload_font(&font8x8);

        // ---- bg tile patterns ----
        make_tile(T_GRASS,  C_GREEN, C_DGREEN);
        make_tile(T_WATER,  C_BLUE,  C_BLUE);
        make_tile(T_FLOWER, C_GREEN, C_YELLOW);

        build_world();
        build_flags();
        sys_ppu_upload(PPU_TILEMAP, &world, 1024);   // push the whole window once (world == window)

        // ---- init ----
        hx = 64; hy = 64; face = 0; flip = 0;
        dialog_on = 0; prev_b = 0; flash_t = 0; flash_restore = 0; prev_a = 0;
        obj_x[1] = 120; obj_y[1] = 40;  obj_pat[1] = S_TREE1; obj_attr[1] = 0;
        obj_x[2] = 200; obj_y[2] = 96;  obj_pat[2] = S_TREE2; obj_attr[2] = 0;
        obj_x[3] = 96;  obj_y[3] = 150; obj_pat[3] = S_TREE3; obj_attr[3] = 0;
        obj_x[4] = 260; obj_y[4] = 64;  obj_pat[4] = S_TREE4; obj_attr[4] = 0;

        while true {
            input = read_input();
            a_now = input & 0x10;
            b_now = input & 0x20;

            // dialog toggle on B (edge-triggered)
            if b_now != 0 {
                if prev_b == 0 {
                    if dialog_on == 0 { dialog_on = 1; } else { dialog_on = 0; }
                }
            }
            prev_b = b_now;

            // palette flash on A (edge-triggered)
            if a_now != 0 {
                if prev_a == 0 { flash_t = 6; flash_restore = 1; }
            }
            prev_a = a_now;

            // movement (frozen while a dialog is open)
            dx = 0; dy = 0;
            if dialog_on == 0 {
                if (input & 0x01) != 0 { dx = 0 - 2; face = 2; flip = 1; }
                if (input & 0x04) != 0 { dx = 2;     face = 2; flip = 0; }
                if (input & 0x02) != 0 { dy = 0 - 2; face = 1; }
                if (input & 0x08) != 0 { dy = 2;     face = 0; }
            }
            moving = 0;
            if dx != 0 { moving = 1; }
            if dy != 0 { moving = 1; }

            nx = hx + dx;
            if blocked(nx, hy) == 0 { hx = nx; }
            ny = hy + dy;
            if blocked(hx, ny) == 0 { hy = ny; }
            if hx < 0 { hx = 0; }
            if hx > 495 { hx = 495; }
            if hy < 0 { hy = 0; }
            if hy > 495 { hy = 495; }

            // animation frame (walk cycle by wall clock; idle -> frame 0)
            now = sys_time();
            frame = now / 150;
            frame = frame & 3;
            if moving == 0 { frame = 0; }
            if face == 1 { base = S_HERO_UP; }
            else { if face == 2 { base = S_HERO_SIDE; } else { base = S_HERO_DOWN; } }
            obj_pat[0] = base + frame;
            obj_x[0] = hx; obj_y[0] = hy; obj_attr[0] = 0;
            if flip == 1 { if face == 2 { obj_attr[0] = 1; } }

            // camera follow, clamped to the world
            camx = hx - 72;
            camy = hy - 56;
            if camx < 0 { camx = 0; }
            if camx > 352 { camx = 352; }
            if camy < 0 { camy = 0; }
            if camy > 384 { camy = 384; }

            // ---- build the frame ----
            cmd_reset();
            ppu_scroll(camx, camy);

            // palette flash: whiten a few entries for a few frames, then DMA-restore
            if flash_t > 0 {
                ppu_pal1(C_GREEN, 0xFFFF);
                ppu_pal1(C_BLUE, 0xFFFF);
                ppu_pal1(7, 0xFFFF);
                flash_t = flash_t - 1;
            } else {
                if flash_restore == 1 {
                    sys_ppu_dma(PAL_MASTER, PPU_PAL);
                    flash_restore = 0;
                }
            }

            // Y-sort the objects by world y (insertion sort of indices), then emit back-to-front.
            i = 0;
            while i < NOBJ { order[i] = i; i = i + 1; }
            i = 1;
            while i < NOBJ {
                j = i;
                while j > 0 {
                    if obj_y[order[j - 1]] > obj_y[order[j]] {
                        t = order[j - 1]; order[j - 1] = order[j]; order[j] = t;
                        j = j - 1;
                    } else { break; }
                }
                i = i + 1;
            }
            i = 0;
            while i < NOBJ {
                oi = order[i];
                oam_set(obj_pat[oi], obj_x[oi] - camx, obj_y[oi] - camy, obj_attr[oi]);
                i = i + 1;
            }

            // HUD (fixed text cells)
            tile_text(0, 0, "OVERWORLD", C_WHITE);

            // dialog box (button B)
            if dialog_on == 1 {
                tile_fill(1, 11, 18, 4, GLYPH_SOLID, C_BLACK);
                tile_text(2, 12, "A HERO WALKS THE", C_WHITE);
                tile_text(2, 13, "QUIET GREEN FIELDS.", C_WHITE);
            }

            ppu_present();
        }
        return 0;
    }
}
