// examples/menu.txt — ROM selector menu (PPU text-plane version).
//
// A normal ROM compiled through the pipeline. It lists the games the host exposes (over
// the syscall/interrupt mechanism) and launches the highlighted one. Boots first on the
// handheld; on the desktop emulator run it with `pc_emu --menu`.
//
// Doubles as a syntax reference for the language: globals, arrays, address-of, function
// calls, while loops, nested if/else, comparisons and bitwise tests.
//
// Renders entirely on the PPU's text plane (no tilemap, no sprites): an opaque black panel
// covers the whole 20x16 grid once at boot (tile_fill with GLYPH_SOLID -- its bitmap is fully
// "on", so even with bg=NO_BG every pixel still paints, giving an opaque block), then each row's
// letters are redrawn ONLY when the highlighted row changes. tile_text_bg's cells PERSIST in PPU
// RAM between frames (compose_text just keeps redrawing whatever's already there), so an
// unchanged frame safely re-presents the same picture for free -- this used to be a hard
// requirement on the old VRAM path (a full redraw couldn't finish inside one frame's instruction
// budget); on the PPU it's just a cheap, still-worth-doing optimization.
//
// Unlike a game ROM, the menu can't assume a .pak is ever present (it's the very first thing
// that boots, before any game's assets exist on the device) -- so it uploads its own tiny
// 3-color palette directly from a baked-in byte array via sys_ppu_upload, instead of DMA-ing one
// from storage like every other PPU program in this repo does.

include "../lib/io.lib";              // read_input
include "../lib/sys.lib";             // sys_list_roms / sys_get_rom_name / sys_load_rom
include "../lib/ppu.lib";             // text plane + the PPU command/present protocol (also font8x8)

{
    // GET_ROM_NAME writes one name here per row. LIST_ROMS (called once, for the count)
    // also dumps ALL names here, so the buffer is sized for the whole list, not one name.
    var byte namebuf[512];

    // A minimal palette: index 0 = black (panel + letter background), 1 = white (unselected),
    // 2 = red (selected) -- RGB565, little-endian, matching sys_ppu_upload's raw byte copy.
    var byte menu_pal[6] = { 0x00, 0x00,   0xFF, 0xFF,   0x00, 0xF8 };
    const IDX_BLACK = 0; const IDX_WHITE = 1; const IDX_RED = 2;

    // Row text starts at column 1 (a small left margin) and the grid is 20 columns wide, so a
    // name longer than 19 chars would run past the row -- the PPU's TEXT command does NOT clamp
    // that (same as the old VRAM plot path), it would just spill into the NEXT row's cells (or
    // past the end of PPU_TEXTMAP entirely for a long enough name on the last row). ROM filenames
    // are short in practice, but they come from whatever's on the storage device, so clamp them.
    const ROW_X = 1; const MAX_NAME = 19;

    int main() {
        var int count;
        var int shown;
        var int selected;
        var int drawn;
        var int prev;
        var int i;
        var int len;
        var int input;
        var int color;

        count = sys_list_roms(&namebuf, 16);    // R0 = number of ROMs

        // MAX_ROMS (every host's own list cap) is 16, which also happens to be exactly the text
        // grid's row count -- so every possible ROM fits with no separate screen-height cap needed
        // (unlike the old VRAM version, which had to cap at 12 rows of 8px text on a 128px screen).
        shown = count;
        if shown > 16 {
            shown = 16;
        }

        selected = 0;
        prev = 0;
        drawn = -1;                             // -1 != selected, so the first frame draws

        sys_ppu_upload(PPU_PAL, &menu_pal, 6);
        ppu_upload_font(&font8x8);
        ppu_scroll(0, 0);

        while (true) {
            cmd_reset();

            // Redraw ONLY when the highlight moved (see the file header for why this is safe).
            if selected != drawn {
                if drawn < 0 {
                    tile_fill(0, 0, 20, 16, GLYPH_SOLID, IDX_BLACK);   // opaque panel, once
                }
                i = 0;
                while (i < shown) {
                    len = sys_get_rom_name(i, &namebuf);
                    if len > MAX_NAME {
                        namebuf[MAX_NAME] = 0;      // clamp: NUL-terminate early, don't overrun the row
                    }
                    if i == selected {
                        color = IDX_RED;            // red (selected)
                    } else {
                        color = IDX_WHITE;           // white
                    }
                    tile_text_bg(ROW_X, i, &namebuf, color, IDX_BLACK);
                    i = i + 1;
                }
                drawn = selected;
            }

            ppu_present();

            // Navigation: act only on a FRESH press (bit set now AND clear last frame).
            // The language has no '!' operator, so test the previous state with '== 0'.
            input = read_input();

            if (input & 8) != 0 {               // DOWN (0x08)
                if (prev & 8) == 0 {
                    if selected < shown - 1 {
                        selected = selected + 1;
                    }
                }
            }
            if (input & 2) != 0 {               // UP (0x02)
                if (prev & 2) == 0 {
                    if selected > 0 {
                        selected = selected - 1;
                    }
                }
            }
            if (input & 16) != 0 {              // A (0x10) -> launch the highlighted ROM
                if (prev & 16) == 0 {
                    sys_load_rom(selected);
                }
            }

            prev = input;
        }

        return 0;
    }
}
