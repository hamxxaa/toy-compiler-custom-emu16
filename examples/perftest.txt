// perftest.txt — exercises the whole draw pipeline so the firmware's per-phase timing shows
// every cost. Each frame it clears the screen (full VRAM change -> dirty-check fires, frame
// buffer rebuilds, screen pushes), moves 10 sprites, and cycles a palette entry (PRAM change
// -> palette rebuilds). Calls sys_present so it's paced like a real game. Read the timing
// breakdown on serial; compare with bench.rom (pure compute, no draw).
include "../lib/io.lib";
include "../lib/sys.lib";
{
    var byte spr[4] = {0xF0, 0xF0, 0xF0, 0xF0};   // 4x4 block

    int main() {
        var int f;
        var int i;
        f = 0;
        while (true) {
            f = f + 1;
            fill(0);                                          // clear -> full VRAM change
            i = 0;
            while (i < 10) {
                draw_sprite(&spr, 4, 4, (f + i * 13) & 127, i * 12, (i & 7) + 1);
                i = i + 1;
            }
            set_palette(1, f << 5);                           // cycle a colour -> PRAM change
            sys_present();
        }
        return 0;
    }
}
