// ─────────────────────────────────────────────────────────────────────────────────────────────
// PPU bigworld — Phase 4 streaming reference. The world (64x48 tiles = 1024x768 px) is far larger
// than the PPU's 32x32 tilemap window, so lib/scene.lib streams the visible tiles into the torus
// each frame while the camera follows the hero. Same asset pak as ppu_overworld.
//
// Build + run:
//   python main.py examples/ppu_bigworld.txt --save-rom ppu_bigworld
//   python tools/image_import.py assets/sprites/sprites.list build/demo_assets
//   python tools/pack_assets.py build/demo_assets/assets.manifest build/roms/ppu_bigworld
//   browser: drop ppu_bigworld.rom + ppu_bigworld.pak.  Arrows move.
// ─────────────────────────────────────────────────────────────────────────────────────────────
include "../lib/io.lib";       // read_input
include "../lib/sys.lib";      // sys_time
include "../lib/scene.lib";    // camera + streaming + Y-sort cast (pulls ppu.lib -- also font8x8)
include "../lib/map.lib";      // map_solid_at (tile collision against the compiled-in world)

{
    const PAL_MASTER  = 0;
    const HERO_SIDE_A = 1;
    const HERO_UP_A   = 3;
    const HERO_DOWN_A = 4;
    const TREE1_A     = 8;

    const S_HERO_SIDE = 0;
    const S_HERO_UP   = 4;
    const S_HERO_DOWN = 8;
    const S_TREE      = 12;
    const T_GRASS     = 16;
    const T_WATER     = 17;   // solid
    const T_FLOWER    = 18;

    const C_GREEN  = 10;
    const C_DGREEN = 12;
    const C_BLUE   = 18;
    const C_YELLOW = 8;
    const C_WHITE  = 21;

    const MAP_W = 64;
    const MAP_H = 48;
    var byte world[3072];      // 64*48; resident world map (also the collision map)
    var byte tile_flags[19];
    var byte tilebuf[256];

    const N_TREES = 6;
    var int tree_x[6] = { 160, 520, 300, 800, 96,  640 };
    var int tree_y[6] = { 120, 200, 400, 300, 500, 96  };

    var int hx; var int hy; var int face; var int flip; var int moving;

    void make_tile(int slot, int base, int speckle) {
        var int i; var int k;
        i = 0;
        while i < 256 {
            k = i & 31;
            if k == 5 { tilebuf[i] = speckle; }
            else { if k == 22 { tilebuf[i] = speckle; } else { tilebuf[i] = base; } }
            i = i + 1;
        }
        sys_ppu_upload(PPU_PAT + slot * 256, &tilebuf, 256);
    }

    // Fill grass in one flat pass, then overwrite borders/lakes/flowers. (A per-cell border check on
    // all 3072 cells would blow the ~100k first-frame instruction budget before the first present.)
    void build_world() {
        var int i; var int r; var int c;
        i = 0;
        while i < 3072 { world[i] = T_GRASS; i = i + 1; }
        c = 0;
        while c < MAP_W {
            world[c] = T_WATER;                        // top row
            world[(MAP_H - 1) * MAP_W + c] = T_WATER;  // bottom row
            c = c + 1;
        }
        r = 0;
        while r < MAP_H {
            world[r * MAP_W] = T_WATER;                // left col
            world[r * MAP_W + MAP_W - 1] = T_WATER;    // right col
            r = r + 1;
        }
        r = 10; while r < 16 { c = 20; while c < 30 { world[r * MAP_W + c] = T_WATER; c = c + 1; } r = r + 1; }
        r = 30; while r < 40 { c = 40; while c < 52 { world[r * MAP_W + c] = T_WATER; c = c + 1; } r = r + 1; }
        r = 4;  while r < 8  { c = 4;  while c < 10 { world[r * MAP_W + c] = T_FLOWER; c = c + 1; } r = r + 1; }
    }

    void build_flags() {
        var int i;
        i = 0; while i < 19 { tile_flags[i] = 0; i = i + 1; }
        tile_flags[T_WATER] = 1;
    }

    int blocked(int px, int py) {
        if map_solid_at(&world, MAP_W, &tile_flags, px + 3,  py + 10) == 1 { return 1; }
        if map_solid_at(&world, MAP_W, &tile_flags, px + 12, py + 10) == 1 { return 1; }
        if map_solid_at(&world, MAP_W, &tile_flags, px + 3,  py + 15) == 1 { return 1; }
        if map_solid_at(&world, MAP_W, &tile_flags, px + 12, py + 15) == 1 { return 1; }
        return 0;
    }

    int main() {
        var int input; var int dx; var int dy; var int nx; var int ny;
        var int now; var int frame; var int base; var int i;

        sys_ppu_dma(PAL_MASTER, PPU_PAL);
        sys_ppu_dma(HERO_SIDE_A, PPU_PAT + S_HERO_SIDE * 256);
        sys_ppu_dma(HERO_UP_A,   PPU_PAT + S_HERO_UP * 256);
        sys_ppu_dma(HERO_DOWN_A, PPU_PAT + S_HERO_DOWN * 256);
        sys_ppu_dma(TREE1_A,     PPU_PAT + S_TREE * 256);
        ppu_upload_font(&font8x8);
        make_tile(T_GRASS,  C_GREEN, C_DGREEN);
        make_tile(T_WATER,  C_BLUE,  C_BLUE);
        make_tile(T_FLOWER, C_GREEN, C_YELLOW);
        build_world();
        build_flags();

        hx = 80; hy = 80; face = 0; flip = 0;

        while true {
            input = read_input();
            dx = 0; dy = 0;
            if (input & 0x01) != 0 { dx = 0 - 2; face = 2; flip = 1; }
            if (input & 0x04) != 0 { dx = 2;     face = 2; flip = 0; }
            if (input & 0x02) != 0 { dy = 0 - 2; face = 1; }
            if (input & 0x08) != 0 { dy = 2;     face = 0; }
            moving = 0;
            if dx != 0 { moving = 1; }
            if dy != 0 { moving = 1; }

            nx = hx + dx;
            if blocked(nx, hy) == 0 { hx = nx; }
            ny = hy + dy;
            if blocked(hx, ny) == 0 { hy = ny; }

            now = sys_time();
            frame = now / 150;
            frame = frame & 3;
            if moving == 0 { frame = 0; }
            if face == 1 { base = S_HERO_UP; }
            else { if face == 2 { base = S_HERO_SIDE; } else { base = S_HERO_DOWN; } }

            scene_follow(hx, hy, MAP_W, MAP_H);

            cmd_reset();
            scene_stream(&world, MAP_W, MAP_H);

            scene_clear();
            i = 0;
            while i < N_TREES {
                scene_obj(S_TREE, tree_x[i], tree_y[i], 0);
                i = i + 1;
            }
            if flip == 1 { if face == 2 { scene_obj(base + frame, hx, hy, 1); } else { scene_obj(base + frame, hx, hy, 0); } }
            else { scene_obj(base + frame, hx, hy, 0); }
            scene_draw();

            tile_text(0, 0, "BIGWORLD 64X48", C_WHITE);
            tile_text(0, 1, "X", C_WHITE);
            tile_number(2, 1, hx, C_WHITE);
            tile_text(8, 1, "Y", C_WHITE);
            tile_number(10, 1, hy, C_WHITE);

            ppu_present();
        }
        return 0;
    }
}
