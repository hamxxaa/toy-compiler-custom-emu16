// bench.txt — a calculation-heavy ROM for measuring CPU throughput. Not a game: it just
// grinds a representative mix of integer math, array load/stores, and function calls in an
// infinite loop. It never calls sys_present, so the host runs a full 100,000-instruction
// slice every frame -> each firmware frame is ~100k instructions of pure compute. The screen
// stays black; read the average frame time on serial. Power-cycle to exit.
include "../lib/game.lib";
{
    var int scratch[16];

    int mix(int v) {
        return (v * 5 + 3) ^ (v >> 2);
    }

    int main() {
        var int a;
        var int i;
        a = 12345;
        while (true) {
            i = a & 15;
            a = a * 7 + 13;
            a = a ^ scratch[i];
            scratch[i] = a;
            a = a / 3;
            a = a - 99;
            a = a + aabb(a & 31, 0, 8, 8, 10, 0, 8, 8);   // call into game.lib
            a = mix(a);                                    // local call
            a = a & 0x7FFF;
        }
        return 0;
    }
}
