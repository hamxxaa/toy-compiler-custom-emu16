{
    var int gCounter = 0;

    int identity(int x) {
        return x;
    }

    int add(int a, int b) {
        return a + b;
    }

    int sub(int a, int b) {
        return a - b;
    }

    int sum3(int a, int b, int c) {
        var int ab = a + b;
        return ab + c;
    }

    int sum4(int a, int b, int c, int d) {
        var int ab = a + b;
        var int cd = c + d;
        return ab + cd;
    }

    int sum5(int a, int b, int c, int d, int e) {
        var int abc = a + b;
        abc = abc + c;
        var int de = d + e;
        return abc + de;
    }

    int nested_call() {
        var int x = add(3, 4);
        var int y = add(10, x);
        return y;
    }

    int deep_nesting() {
        return add(add(1, 2), add(3, add(4, 5)));
    }

    int conditional(int x) {
        var int result = 0;
        if x > 10 do {
            result = 1;
        }
        if x < 5 do {
            result = 2;
        }
        if x == 7 do {
            result = 3;
        }
        return result;
    }

    int loop_sum(int n) {
        var int i = 0;
        var int total = 0;
        while i < n do {
            total = total + i;
            i = i + 1;
        }
        return total;
    }

    int countdown(int n) {
        var int result = 0;
        while n > 0 do {
            result = result + n;
            n = n - 1;
        }
        return result;
    }

    int fibonacci(int n) {
        var int a = 0;
        var int b = 1;
        var int i = 0;
        while i < n do {
            var int temp = b;
            b = a + b;
            a = temp;
            i = i + 1;
        }
        return a;
    }

    int use_global() {
        gCounter = gCounter + 10;
        return gCounter;
    }

    int multi_comparison(int x) {
        var int r = 0;
        if x >= 10 do {
            r = r + 1;
        }
        if x <= 5 do {
            r = r + 10;
        }
        if x != 0 do {
            r = r + 100;
        }
        return r;
    }

    int reassignment() {
        var int x = 1;
        x = x + 1;
        x = x + 1;
        x = x + 1;
        x = x + 1;
        return x;
    }

    int call_chain() {
        var int a = identity(5);
        var int b = add(a, a);
        var int c = sub(b, 3);
        var int d = add(c, identity(1));
        return d;
    }

    int is_even(int n) {
        var int half = 0;
        var int doubled = 0;
        while doubled < n do {
            half = half + 1;
            doubled = half + half;
        }
        if doubled == n do {
            return 1;
        }
        return 0;
    }

    int count_evens(int limit) {
        var int i = 0;
        var int count = 0;
        while i < limit do {
            var int e = is_even(i);
            if e == 1 do {
                count = count + 1;
            }
            i = i + 1;
        }
        return count;
    }

    int abs_diff(int a, int b) {
        if a > b do {
            return a - b;
        }
        return b - a;
    }

    int multi_return(int x) {
        if x < 0 do {
            return 0 - x;
        }
        if x == 0 do {
            return 0;
        }
        return x;
    }

        if t12 != 10 do { fail = fail + 1; }
