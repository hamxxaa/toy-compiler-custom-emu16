// block_blast — brick-breaker ported to the PPU model (from the pre-PPU VRAM original; see git
// history for that version). Fully self-contained: no .pak, all art is baked directly into the ROM
// as tiny hand-computed 16x16 patterns (a circle for the ball/aim-dot, bordered boxes for bricks).
//
// Screen (160x128 = 10 cols x 8 rows of 16px tiles):
//   row 0        HUD (text plane only; tile background stays blank)
//   rows 1-6     brick playfield, 10x6 = 60 cells
//   row 7        launcher zone (ball spawn + aim-trajectory preview)
// Bricks are BACKGROUND TILES (not sprites) -- avoids the 64-slot OAM budget entirely regardless of
// grid size. Each brick's HP number is drawn via the text plane, transparent background, in the top
// 2x2 text-cells of its tile (a tile is exactly 2x2 text-cells, since text cells are 8px and tiles
// are 16px). The whole grid + HUD is rebuilt from scratch every frame (cheap: ~60 tiles + a few text
// writes, well inside the 2 KB command budget) -- unlike the original, there's no dirty-tracking,
// since PPU command-buffer building doesn't need it the way the old VRAM path did.
//
// Balls (up to MAX_BALLS) and the aim-trajectory preview dots are both OAM sprites, but never at the
// same time: dots only exist during aim (gameState 0, no balls in flight yet), balls only exist
// during flight (gameState 1). MAX_BALLS is capped below the original's 49 purely as a safety margin
// under the 64-sprite OAM budget, not because they'd ever coexist with dots in practice.

    // ── Grid ─────────────────────────────────────────────────────────────────────────────────────
    const MAP_W    = 10;
    const MAP_H    = 6;
    const CELL     = 16;
    const GRID_Y0  = 16;    // pixel y of the playfield's top edge (row 0 is reserved for the HUD)

    var int aimy[31] = { -66,-88,-108,-128,-147,-165,-181,-196,-210,-222,-232,-241,-247,-252,-255,-256,-255,-252,-247,-241,-232,-222,-210,-196,-181,-165,-147,-128,-108,-88,-66 };

    int cell_brick(int px, int py) {
        var int col; var int row;
        if py < GRID_Y0 { return 0; }
        col = px / CELL;
        row = (py - GRID_Y0) / CELL;
        if col < 0 { return 0; }
        if col >= MAP_W { return 0; }
        if row >= MAP_H { return 0; }
        return bricks[row * MAP_W + col];
    }

    int active_count() {
        var int b = 0;
        var int c = 0;
        while b < MAX_BALLS {
            if ball_on[b] == 1 { c = c + 1; }
            b = b + 1;
        }
        return c;
    }

    int tier_pattern(int hp) {
        if hp <= 0 { return 0; }
        if hp < 4  { return PAT_TIER1; }
        if hp < 8  { return PAT_TIER2; }
        if hp < 16 { return PAT_TIER3; }
        return PAT_TIER4;
    }

    // Rebuild the whole brick tilemap + per-brick HP numbers from bricks[]. Cheap enough (60 tiles)
    // to just do every frame -- no dirty-tracking needed on the PPU model.
    int draw_bricks() {
        var int r = 0;
        var int c;
        var int hp;
        var int tx;
        var int ty;
        while r < MAP_H {
            c = 0;
            while c < MAP_W {
                hp = bricks[r * MAP_W + c];
                tx = c;
                ty = r + 1;                     // +1: row 0 is the HUD
                cu8(3); cu8(tx); cu8(ty); cu8(1); cu8(1); cu8(tier_pattern(hp));
                // Always clear the 2-wide digit row first: tile_number only writes as many cells as
                // the new number's digit count, so a HP drop from 2 digits to 1 (e.g. 10 -> 9) would
                // otherwise leave the old tens digit's cell untouched (reads as "90").
                tile_clear(tx * 2, ty * 2, 2, 1);
                if hp > 0 {
                    tile_number(tx * 2, ty * 2, hp, 1);
                }
                c = c + 1;
            }
            r = r + 1;
        }
        return 0;
    }

    int draw_hud() {
        tile_text(0, 0, "B:", 1);
        tile_number(2, 0, ball_count, 1);
        tile_text(12, 0, "S:", 1);
        tile_number(14, 0, destroyed, 1);
        return 0;
    }

    // Simulate the shot's bounce physics (walls + bricks) and record up to MAX_DOTS sample points
    // along the path. Only called during gameState 0, when no balls are on screen yet.
    int compute_aim_dots() {
        var int tx; var int ty; var int tvx; var int tvy;
        var int done; var int count; var int px; var int py;
        tx = ball_shootingx << 7;
        ty = ball_shootingy << 7;
        tvx = aimx[aim];
        tvy = aimy[aim];
        done = 0; count = 0; dot_n = 0;
        while done == 0 {
            tx = tx + tvx;
            ty = ty + tvy;
            if tx < 0     { tx = 0;     tvx = 0 - tvx; }
            if tx > 20352 { tx = 20352; tvx = 0 - tvx; }
            if ty < 2048  { ty = 2048;  tvy = 0 - tvy; }
            px = tx >> 7;
            py = ty >> 7;
            if py >= 127 { done = 1; }
            if done == 0 {
                if cell_brick(px, py) != 0 { done = 1; }
            }
            if done == 0 {
                if (count & 15) == 0 {
                    if dot_n < MAX_DOTS {
                        dotx[dot_n] = px;
                        doty[dot_n] = py;
                        dot_n = dot_n + 1;
                    }
                }
            }
            count = count + 1;
            if count > 300 { done = 1; }
        }
        return 0;
    }

    int spawn_row() {
        var int c;
        var int base_hp = 1 + level / 3;
        var int hp;
        var int all_full;
        var int all_empty;
        var int fc;
        c = 0;
        while c < MAP_W {
            if rand_range(9) < 5 {
                hp = base_hp + rand_range(2);
                bricks[c] = hp;
            } else {
                bricks[c] = 0;
            }
            c = c + 1;
        }
        all_full = 1;
        c = 0;
        while c < MAP_W {
            if bricks[c] == 0 { all_full = 0; }
            c = c + 1;
        }
        if all_full == 1 {
            fc = rand_range(MAP_W);
            bricks[fc] = 0;
        }
        all_empty = 1;
        c = 0;
        while c < MAP_W {
            if bricks[c] != 0 { all_empty = 0; }
            c = c + 1;
        }
        if all_empty == 1 {
            fc = rand_range(MAP_W);
            bricks[fc] = base_hp;
        }
        return 0;
    }

    int advance_level() {
        var int r; var int c;
        var int gameover = 0;
        c = 0;
        while c < MAP_W {
            if bricks[(MAP_H - 1) * MAP_W + c] != 0 { gameover = 1; }
            c = c + 1;
        }
        if gameover == 1 {
            gameState = 2;
            return 0;
        }
        r = MAP_H - 1;
        while r >= 1 {
            c = 0;
            while c < MAP_W {
                bricks[r * MAP_W + c] = bricks[(r - 1) * MAP_W + c];
                c = c + 1;
            }
            r = r - 1;
        }
        c = 0;
        while c < MAP_W {
            bricks[c] = 0;
            c = c + 1;
        }
        spawn_row();
        while destroyed >= next_ball_at {
            ball_count   = ball_count + 1;
            next_ball_at = next_ball_at + 10;
        }
        if ball_count > MAX_BALLS { ball_count = MAX_BALLS; }
        level = level + 1;
        return 0;
    }

    int reset_game() {
        var int i = 0;
        var int b = 0;
        while i < 60 {
            bricks[i] = 0;
            i = i + 1;
        }
        level          = 0;
        destroyed      = 0;
        ball_count     = 3;
        next_ball_at   = 10;
        ball_shootingx = 80;
        next_x         = 80;
        landed         = 0;
        aim            = 15;
        balls_left     = 0;
        launch_timer   = 0;
        while b < MAX_BALLS {
            ball_on[b] = 0;
            b = b + 1;
        }
        spawn_row();
        gameState = 0;
        return 0;
    }

    // ---------------------------------------------------------------
    // Ball physics + draw (no erase pass needed -- bricks are redrawn fresh every frame, and balls
    // are separate OAM sprites composited on top, not pixels baked into a shared buffer)
    // ---------------------------------------------------------------

    int update_balls() {
        var int b = 0;
        var int s; var int steps;
        var int opx; var int opy; var int nx; var int ny;
        var int col; var int row; var int idx;
        var int oldcol; var int oldrow;
        steps = speed * ff;

        while b < MAX_BALLS {
            if ball_on[b] == 1 {
                s = 0;
                while s < steps {
                    opx = ball_px[b];
                    opy = ball_py[b];

                    ball_px[b] = ball_px[b] + ball_vx[b];
                    ball_py[b] = ball_py[b] + ball_vy[b];

                    if ball_px[b] < 0     { ball_px[b] = 0;     ball_vx[b] = 0 - ball_vx[b]; }
                    if ball_px[b] > 20352 { ball_px[b] = 20352; ball_vx[b] = 0 - ball_vx[b]; }
                    if ball_py[b] < 2048  { ball_py[b] = 2048;  ball_vy[b] = 0 - ball_vy[b]; }

                    nx = ball_px[b] >> 7;
                    ny = ball_py[b] >> 7;

                    if ny >= 127 {
                        ball_on[b] = 0;
                        if landed == 0 { next_x = nx; landed = 1; }
                        s = steps;
                    } else {
                        col = nx / CELL;
                        if col < MAP_W {
                            if ny >= GRID_Y0 {
                                row = (ny - GRID_Y0) / CELL;
                                if row < MAP_H {
                                    idx = row * MAP_W + col;
                                    if bricks[idx] != 0 {
                                        bricks[idx] = bricks[idx] - 1;
                                        if bricks[idx] == 0 { destroyed = destroyed + 1; }
                                        oldcol = (opx >> 7) / CELL;
                                        if (opy >> 7) < GRID_Y0 {
                                            oldrow = 0 - 1;
                                        } else {
                                            oldrow = ((opy >> 7) - GRID_Y0) / CELL;
                                        }
                                        if oldrow != row { ball_vy[b] = 0 - ball_vy[b]; }
                                        if oldcol != col { ball_vx[b] = 0 - ball_vx[b]; }
                                        ball_px[b] = opx;
                                        ball_py[b] = opy;
                                    }
                                }
                            }
                        }
                        s = s + 1;
                    }
                }
                if ball_on[b] == 1 {
                    nx = ball_px[b] >> 7;
                    ny = ball_py[b] >> 7;
                    oam_set(PAT_BALL, nx - 8, ny - 8, 0);
                }
            }
            b = b + 1;
        }
        return 0;
    }

        sys_ppu_upload(PPU_PAL, &game_pal, 16);
        ppu_upload_font(&font8x8);
        sys_ppu_upload(PPU_PAT + PAT_BALL  * 256, &pat_ball,  256);
        sys_ppu_upload(PPU_PAT + PAT_DOT   * 256, &pat_dot,   256);
        sys_ppu_upload(PPU_PAT + PAT_TIER1 * 256, &pat_tier1, 256);
        sys_ppu_upload(PPU_PAT + PAT_TIER2 * 256, &pat_tier2, 256);
        sys_ppu_upload(PPU_PAT + PAT_TIER3 * 256, &pat_tier3, 256);
        sys_ppu_upload(PPU_PAT + PAT_TIER4 * 256, &pat_tier4, 256);

        srand_time();

        while b < MAX_BALLS {
            ball_on[b] = 0;
            b = b + 1;
        }
        spawn_row();

        while (true) {
            cmd_reset();
            ppu_backdrop(0x0000);

            if gameState == 0 {
                // ---- Aim ----
                if button_pressed(BTN_LEFT)  { if aim > 0  { aim = aim - 1; } }
                if button_pressed(BTN_RIGHT) { if aim < 30 { aim = aim + 1; } }

                draw_bricks();
                draw_hud();
                compute_aim_dots();
                oam_set(PAT_BALL, ball_shootingx - 8, ball_shootingy - 8, 0);
                i = 0;
                while i < dot_n {
                    oam_set(PAT_DOT, dotx[i] - 8, doty[i] - 8, 0);
                    i = i + 1;
                }

                if button_pressed(BTN_A) {
                    shot_vx      = aimx[aim];
                    shot_vy      = aimy[aim];
                    balls_left   = ball_count;
                    launch_timer = 0;
                    landed       = 0;
                    ff           = 1;
                    gameState    = 1;
                }

            } else if gameState == 1 {
                // ---- Flight ----
                if button_pressed(BTN_B) {
                    b = 0;
                    while b < MAX_BALLS {
                        ball_on[b] = 0;
                        b = b + 1;
                    }
                    balls_left = 0;
                }

                ff = 1;
                if button_pressed(BTN_X) { ff = 3; }

                if balls_left > 0 {
                    if launch_timer == 0 {
                        slot = ball_count - balls_left;
                        ball_px[slot] = ball_shootingx << 7;
                        ball_py[slot] = ball_shootingy << 7;
                        ball_vx[slot] = shot_vx;
                        ball_vy[slot] = shot_vy;
                        ball_on[slot] = 1;
                        balls_left    = balls_left - 1;
                        launch_timer  = 4;
                    } else {
                        launch_timer = launch_timer - 1;
                    }
                }

                draw_bricks();
                draw_hud();
                update_balls();

                if balls_left == 0 {
                    if active_count() == 0 {
                        if landed == 1 {
                            ball_shootingx = next_x;
                            if ball_shootingx < 8   { ball_shootingx = 8;   }
                            if ball_shootingx > 151 { ball_shootingx = 151; }
                        }
                        advance_level();
                        // advance_level() only sets gameState=2 on game over; otherwise it must
                        // come back here so the next frame goes to aim, not straight back into
                        // this same balls_left==0/active_count()==0 branch again (which would call
                        // advance_level() again on EVERY subsequent frame -- shifting the grid down
                        // dozens of times a second instead of once per volley).
                        if gameState != 2 { gameState = 0; }
                    }
                }

            } else {
                // ---- Game Over ----
                if button_pressed(BTN_A) {
                    // One-time clear on the transition frame: the text plane holds whatever was
                    // last drawn to each cell until something overwrites it, so without this the
                    // "GAME OVER"/"PRESS A" glyphs (and the solid black curtain under them) would
                    // sit there forever once we're back in aim/flight, since neither of those
                    // states happens to touch every cell this screen used.
                    tile_clear(0, 0, 20, 16);
                    reset_game();
                } else {
                    tile_fill(0, 0, 20, 16, GLYPH_SOLID, 0);
                    tile_text_bg(5, 5, "GAME OVER", 1, 0);
                    tile_number(9, 7, destroyed, 1);
                    tile_text_bg(6, 9, "PRESS A", 1, 0);
                }
            }

            buttons_update();
            ppu_present();
        }
        return 0;
    }
