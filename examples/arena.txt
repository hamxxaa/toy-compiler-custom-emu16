// arena -- roam + real combat on the scene engine. Thin GAME on top of scene.lib:
//   * HERO (objs[0]): arrows move (4-dir), A attacks (a one-shot whose strike frame deals damage in
//     a box in front of you). Has HP; enemy contact hurts you (with i-frames + a white hit-flash).
//   * ENEMIES (objs[1..NUM_ENEMIES]): wander, have HP, blink+knockback when hit, despawn on death and
//     respawn after a delay.
//   * HASTA-NINJA (objs[I_NINJA]): a solid NPC. Press B while facing him to talk -- a one-time first
//     line (gated by flag F_NINJA_MET via event.lib), then his catchphrase every time after.
//   * TREES (objs[..]): solid Y-sorted landmarks.
//   Solid ground -> dirty-rect, no full clear. Build assets first:
//     python tools/image_import.py assets/sprites.list build/arena
//     python tools/pack_assets.py build/arena/assets.manifest build/roms/arena
//     cp build/arena/sprites.gen.txt examples/arena_assets.gen.txt
include "arena_assets.gen.txt";       // clip registry
include "../lib/scene.lib";            // engine: cast + camera + render + collision + interact/zones
include "../lib/game.lib";             // rand_range / srand_time
include "../lib/text.lib";             // draw_number for the HUD
include "../lib/event.lib";            // flags + dialog (the ninja's talk event)

{
    const WORLD_W = 480;
    const WORLD_H = 384;
    const GROUND  = 15;

    const I_PLAYER    = 0;
    const NUM_ENEMIES = 4;
    const I_ENEMY0    = 1;            // enemies at objs[1 .. NUM_ENEMIES]
    const I_NINJA     = I_ENEMY0 + NUM_ENEMIES;   // 5: the hasta-ninja NPC, spawned before the trees
    const NUM_TREES   = 14;

    // hasta-ninja NPC: a solid landmark you talk to (press B while facing him)
    const NINJA_X = 100; const NINJA_Y = 60;     // just east of the player's spawn
    const NINJA_REACH = 12;                       // interact box depth in front of the player
    const F_NINJA_MET = 0;                        // event flag: have we heard his first line?

    const DIR_SIDE = 0; const DIR_UP = 1; const DIR_DOWN = 2;

    // combat tuning
    const PLAYER_HP    = 6;
    const ENEMY_HP     = 3;
    const ATTACK_DMG   = 1;
    const ATTACK_REACH = 10;
    const STRIKE_FRAME = 1;           // attack clip frame on which the hit registers
    const HURT_BLINK   = 20;
    const IFRAMES      = 40;
    const KB           = 6;
    const RESPAWN_T    = 120;
    const HP_FILL = 32;              // green (current health)
    const HP_BG   = 1;              // dark panel (empty)

    var int player_dir;
    var int player_iframes;
    var int attack_fired;            // one strike per swing
    var int flash_timer;             // white hit-flash countdown

    var int en_dir[NUM_ENEMIES];
    var int en_timer[NUM_ENEMIES];
    var int en_respawn[NUM_ENEMIES]; // counts down while despawned

    var int fps; var int fps_count; var int fps_last;

    // --- map: {tree_type 0..3, world x, world y} per tree ---
    var int tree_clip[4];
    var int tree_data[42] = {        // NUM_TREES * 3 (sizes must be a literal/const)
        0,  40,  40,    1, 130,  30,    2, 250,  60,    3, 360,  25,
        1,  70, 150,    2, 200, 130,    0, 330, 170,    3, 430, 110,
        2,  30, 260,    0, 150, 300,    3, 280, 250,    1, 400, 290,
        0,  90, 350,    2, 230, 360
    };

    void spawn_trees() {
        var int i; var int t; var int id;
        tree_clip[0] = ANIM_TREE1_ID; tree_clip[1] = ANIM_TREE2_ID;
        tree_clip[2] = ANIM_TREE3_ID; tree_clip[3] = ANIM_TREE4_ID;
        i = 0;
        while i < NUM_TREES {
            t = tree_data[i * 3];
            id = spawn_obj(tree_clip[t], tree_data[i * 3 + 1], tree_data[i * 3 + 2]);
            set_solid(id, 1);
            set_collider(id, 5, 11, 6, 4);
            i = i + 1;
        }
    }

    void respawn_enemy(int e) {
        var int oi;
        oi = I_ENEMY0 + e;
        objs[oi].wx = rand_range(WORLD_W - 16);
        objs[oi].wy = rand_range(WORLD_H - 16);
        objs[oi].hp = ENEMY_HP;
        objs[oi].active = 1;
        objs[oi].blink = 0;
        play_obj(oi, ANIM_ENEMY_WALK_ID);
        en_dir[e] = rand_range(4);
        en_timer[e] = 40 + rand_range(60);
    }

    void player_respawn() {
        objs[I_PLAYER].wx = 60; objs[I_PLAYER].wy = 60;
        objs[I_PLAYER].hp = PLAYER_HP;
        player_iframes = IFRAMES;
        objs[I_PLAYER].blink = IFRAMES;
    }

    void player_update(int attack) {
        var int moving; var int c; var int dx; var int dy;
        tick_obj(I_PLAYER);
        if objs[I_PLAYER].blink > 0 { objs[I_PLAYER].blink = objs[I_PLAYER].blink - 1; }
        if player_iframes > 0 { player_iframes = player_iframes - 1; }
        moving = 0; dx = 0; dy = 0;
        if button(BTN_LEFT)  { dx = 0 - 1; moving = 1; objs[I_PLAYER].facing = 1; player_dir = DIR_SIDE; }
        if button(BTN_RIGHT) { dx = 1;     moving = 1; objs[I_PLAYER].facing = 0; player_dir = DIR_SIDE; }
        if button(BTN_UP)    { dy = 0 - 1; moving = 1; player_dir = DIR_UP; }
        if button(BTN_DOWN)  { dy = 1;     moving = 1; player_dir = DIR_DOWN; }
        move_obj(I_PLAYER, dx, dy);
        if obj_locked(I_PLAYER) == 0 {
            if attack == 1 {
                play_obj_once(I_PLAYER, ANIM_HERO_ATTACK_ID);
            } else {
                if player_dir == DIR_UP        { c = ANIM_HERO_WALK_UP_ID; }
                else if player_dir == DIR_DOWN { c = ANIM_HERO_WALK_DOWN_ID; }
                else                           { c = ANIM_HERO_WALK_ID; }
                play_obj(I_PLAYER, c);
                if moving == 0 { objs[I_PLAYER].start = sys_time(); }   // idle -> hold frame 0
            }
        }
    }

    void enemy_update() {
        var int e; var int oi; var int c; var int dx; var int dy; var int bl;
        e = 0;
        while e < NUM_ENEMIES {
            oi = I_ENEMY0 + e;
            if objs[oi].active == 1 {
                if objs[oi].blink > 0 { objs[oi].blink = objs[oi].blink - 1; }
                tick_obj(oi);
                en_timer[e] = en_timer[e] - 1;
                if en_timer[e] <= 0 {
                    en_dir[e] = rand_range(4);
                    en_timer[e] = 40 + rand_range(60);
                }
                dx = 0; dy = 0;
                if en_dir[e] == 0      { dx = 1;     objs[oi].facing = 0; c = ANIM_ENEMY_WALK_ID; }
                else if en_dir[e] == 1 { dx = 0 - 1; objs[oi].facing = 1; c = ANIM_ENEMY_WALK_ID; }
                else if en_dir[e] == 2 { dy = 0 - 1; objs[oi].facing = 0; c = ANIM_ENEMY_WALK_UP_ID; }
                else                   { dy = 1;     objs[oi].facing = 0; c = ANIM_ENEMY_WALK_DOWN_ID; }
                play_obj(oi, c);
                bl = move_obj(oi, dx, dy);
                if bl == 1 { en_timer[e] = 0; }
                if objs[oi].wx <= 0            { en_dir[e] = 0; }
                if objs[oi].wx >= WORLD_W - 16 { en_dir[e] = 1; }
                if objs[oi].wy <= 0            { en_dir[e] = 3; }
                if objs[oi].wy >= WORLD_H - 16 { en_dir[e] = 2; }
            } else {
                en_respawn[e] = en_respawn[e] - 1;
                if en_respawn[e] <= 0 { respawn_enemy(e); }
            }
            e = e + 1;
        }
    }

    // On the strike frame of the swing, damage every enemy in the hitbox once.
    void resolve_attack() {
        var int af; var int e; var int oi;
        var int hx; var int hy; var int hw; var int hh; var int kdx; var int kdy;
        if obj_locked(I_PLAYER) == 0 { attack_fired = 0; return; }   // not attacking -> rearm
        af = clip_frame(ANIM_HERO_ATTACK_ID, sys_time() - objs[I_PLAYER].start);
        if af < STRIKE_FRAME { return; }
        if attack_fired == 1 { return; }
        attack_fired = 1;

        kdx = 0; kdy = 0;
        if player_dir == DIR_UP {
            hx = objs[I_PLAYER].wx; hy = objs[I_PLAYER].wy - ATTACK_REACH; hw = 16; hh = ATTACK_REACH; kdy = 0 - KB;
        } else if player_dir == DIR_DOWN {
            hx = objs[I_PLAYER].wx; hy = objs[I_PLAYER].wy + 16; hw = 16; hh = ATTACK_REACH; kdy = KB;
        } else if objs[I_PLAYER].facing == 1 {
            hx = objs[I_PLAYER].wx - ATTACK_REACH; hy = objs[I_PLAYER].wy; hw = ATTACK_REACH; hh = 16; kdx = 0 - KB;
        } else {
            hx = objs[I_PLAYER].wx + 16; hy = objs[I_PLAYER].wy; hw = ATTACK_REACH; hh = 16; kdx = KB;
        }

        e = 0;
        while e < NUM_ENEMIES {
            oi = I_ENEMY0 + e;
            if objs[oi].active == 1 {
                if obj_overlap_box(oi, hx, hy, hw, hh) == 1 {
                    objs[oi].hp = objs[oi].hp - ATTACK_DMG;
                    objs[oi].blink = HURT_BLINK;
                    move_obj(oi, kdx, kdy);
                    if objs[oi].hp <= 0 { despawn(oi); en_respawn[e] = RESPAWN_T; }
                }
            }
            e = e + 1;
        }
    }

    void player_hurt_check() {
        var int e; var int oi; var int kdx; var int kdy;
        if player_iframes > 0 { return; }
        e = 0;
        while e < NUM_ENEMIES {
            oi = I_ENEMY0 + e;
            if objs[oi].active == 1 {
                if obj_overlap(I_PLAYER, oi) == 1 {
                    objs[I_PLAYER].hp = objs[I_PLAYER].hp - 1;
                    player_iframes = IFRAMES;
                    objs[I_PLAYER].blink = IFRAMES;
                    kdx = KB; kdy = KB;
                    if objs[I_PLAYER].wx < objs[oi].wx { kdx = 0 - KB; }
                    if objs[I_PLAYER].wy < objs[oi].wy { kdy = 0 - KB; }
                    move_obj(I_PLAYER, kdx, kdy);
                    flash_timer = 3;
                    palette_fill(0, 256, 0xFFFF);                 // white screen flash
                    if objs[I_PLAYER].hp <= 0 { player_respawn(); }
                    e = NUM_ENEMIES;                              // one hit per frame
                }
            }
            e = e + 1;
        }
    }

    // The hasta-ninja's talk event. First time: a one-time line (gated by F_NINJA_MET). Every time
    // after: he just says his catchphrase. Pure event.lib flags + dialog.
    void talk_ninja() {
        if flag_get(F_NINJA_MET) == 0 {
            flag_set(F_NINJA_MET);
            dialog_open("A ninja appears. 'Hasta... you found me!'");
        } else {
            dialog_open("He says: I am a sick ninja.");
        }
    }

    // Route an interacted object id to its event (the O(1) switch dispatch). I_NINJA is a compile-time
    // constant, so this is a real jump-table dispatch ready for more NPCs; other objects are no-ops.
    void dispatch_interact(int obj) {
        switch (obj) {
            case I_NINJA: talk_ninja();
            default: { }
        }
    }

    int main() {
        var int attack; var int e; var int t;

        sys_set_fps(30);
        sys_asset_load(PAL_MASTER, PRAM, 512);
        scene_reset();
        scene_world(WORLD_W, WORLD_H, GROUND);
        scene_deadzone(60, 44, 84, 68);

        anim_load(ANIM_HERO_WALK_ID);   anim_load(ANIM_HERO_ATTACK_ID);
        anim_load(ANIM_HERO_WALK_UP_ID); anim_load(ANIM_HERO_WALK_DOWN_ID);
        anim_load(ANIM_ENEMY_WALK_ID);  anim_load(ANIM_ENEMY_WALK_UP_ID); anim_load(ANIM_ENEMY_WALK_DOWN_ID);
        anim_load(ANIM_TREE1_ID); anim_load(ANIM_TREE2_ID); anim_load(ANIM_TREE3_ID); anim_load(ANIM_TREE4_ID);
        anim_load(ANIM_NINJA_ID);

        srand_time();

        spawn_obj(ANIM_HERO_WALK_ID, 60, 60);            // objs[I_PLAYER]
        set_collider(I_PLAYER, 4, 11, 8, 4);
        objs[I_PLAYER].hp = PLAYER_HP;
        player_dir = DIR_DOWN; player_iframes = 0; attack_fired = 0; flash_timer = 0;

        e = 0;
        while e < NUM_ENEMIES {                          // objs[I_ENEMY0 .. ]
            spawn_obj(ANIM_ENEMY_WALK_ID, 150 + e * 70, 100 + e * 50);
            set_collider(I_ENEMY0 + e, 4, 11, 8, 4);
            objs[I_ENEMY0 + e].hp = ENEMY_HP;
            en_dir[e] = rand_range(4);
            en_timer[e] = 40 + rand_range(60);
            en_respawn[e] = 0;
            e = e + 1;
        }

        spawn_obj(ANIM_NINJA_ID, NINJA_X, NINJA_Y);      // objs[I_NINJA] -- solid NPC you can talk to
        set_solid(I_NINJA, 1);
        set_collider(I_NINJA, 3, 8, 10, 8);

        spawn_trees();

        fps = 0; fps_count = 0; fps_last = sys_time();
        fill(GROUND);

        while (true) {
            if dialog_active() == 1 {
                // World frozen while the box is up; A advances/closes it.
                dialog_update();
                if dialog_active() == 0 {                    // just closed -> repaint the world (clears the box)
                    scene_refresh();
                    camera_follow(objs[I_PLAYER].wx, objs[I_PLAYER].wy);
                    scene_render();
                }
            } else {
                attack = 0;
                if button(BTN_A) { attack = 1; }
                player_update(attack);
                enemy_update();
                resolve_attack();
                player_hurt_check();

                if button_pressed(BTN_B) {                   // press B facing the ninja to talk
                    t = interact_front(I_PLAYER, player_dir, objs[I_PLAYER].facing, NINJA_REACH);
                    if t >= 0 { dispatch_interact(t); }
                }

                camera_follow(objs[I_PLAYER].wx, objs[I_PLAYER].wy);
                scene_render();

                if flash_timer > 0 {
                    flash_timer = flash_timer - 1;
                    if flash_timer == 0 { sys_asset_load(PAL_MASTER, PRAM, 512); }   // restore colors
                }

                fps_count = fps_count + 1;
                if sys_time() - fps_last >= 1000 {
                    fps = fps_count; fps_count = 0; fps_last = sys_time();
                }
                fill_rect(0, 0, 24, 9, 0);                       // FPS panel
                draw_number(fps, 1, 1, 21);
                fill_rect(0, 11, PLAYER_HP * 4 + 2, 5, HP_BG);   // HP bar (empty)
                fill_rect(1, 12, objs[I_PLAYER].hp * 4, 3, HP_FILL);   // HP bar (filled)
            }

            buttons_update();
            sys_present();
        }
        return 0;
    }
}
