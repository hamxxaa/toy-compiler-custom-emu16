// ─────────────────────────────────────────────────────────────────────────────────────────────
// arena (PPU + pak-loaded, multi-map worlds) — the combat arena, now split across TWO maps
// ("arena" and "grove") loaded at runtime from the .pak via map_load (lib/map.lib -- see
// plans/swapping-worldmaps-kojima.md). Each map is painted with tools/pixel_map.py's PNG+legend
// convention, one pixel = one tile, and authored (entries/spawns/warps) via tools/map_set.py.
// Objects (player, slimes, NPC) are placed from each map's own spawn/entry data -- no compiled-in
// world[] anymore, no hardcoded spawn coordinates. The two maps connect via one warp: the arena's
// south edge (tile 20,31) leads to the grove's north edge (tile 20,0), and back.
//
// Build:
//   python tools/image_import.py assets/sprites/sprites.list build/demo_assets
//   cp build/demo_assets/sprites.gen.txt examples/generated/arena_assets.gen.txt
//   python tools/map_set.py assets/maps assets/maps/legend.txt build/demo_assets --name arena \
//       --assets build/demo_assets/assets.manifest
//   cp build/demo_assets/maps.gen.txt examples/generated/arena_maps.gen.txt
//   python main.py examples/arena.txt --save-rom arena
//   grep -v '^#' build/demo_assets/assets.manifest build/demo_assets/maps.manifest \
//       > build/demo_assets/combined.manifest
//   python tools/pack_assets.py build/demo_assets/combined.manifest build/roms/arena
// (ASSET_MAP_ARENA/ASSET_MAP_GROVE below are hardcoded from that combined manifest's line order:
//  the 18 sprite/palette assets first, then the 2 maps -- ids 18 and 19. If sprites.list ever
//  gains/loses an asset, recompute these two from a fresh build's build/roms/arena_assets.txt.)
// ─────────────────────────────────────────────────────────────────────────────────────────────
include "../lib/io.lib";              // read_input + buttons
include "../lib/game.lib";            // rand_range / srand_time / aabb (pulls sys.lib -> sys_time)
include "../lib/event.lib";           // flags (the NPC's one-time line)
include "../lib/scene.lib";           // camera + Y-sort cast (pulls ppu.lib -- also font8x8 now)
include "../lib/map.lib";             // map_load + spawn/warp/entry reads + map_solid_at
include "../lib/anim.lib";            // clip timing -> OAM pattern slot (anim_play / anim_pat)
include "generated/arena_assets.gen.txt";  // ANIM_*_ID / PAL_MASTER sprite asset ids
include "generated/arena_maps.gen.txt";    // SPAWN_* type ids (from tools/map_set.py)

{
    // The two maps' pak ids (see the Build recipe above for how these are derived).
    const ASSET_MAP_ARENA = 18;
    const ASSET_MAP_GROVE = 19;

    // pattern slots: sprites 0..79 (PPU_SPRITE_SLOTS in map.lib); terrain lives at 80+
    // (assets/maps/legend.txt). Sprites avoid the terrain ids the map uses (80..87, 104).
    const S_H_SIDE = 0;  const S_H_ATK = 4; const S_H_UP = 8;  const S_H_DOWN = 12;
    const S_E_SIDE = 24; const S_E_UP = 29; const S_E_DOWN = 34; const S_NPC = 39;
    const ANIM_PERIOD = 200;   // ms per animation frame (arena's clips are uniform; anim.lib is per-clip)
    const T_TREE5 = 104;  // imported, not yet painted into any map -- ready for a future repaint

    const C_WHITE = 21; const C_RED = 27; const C_BLACK = 0; const C_GREEN = 10;

    // object model: 0=player, 1..4=slimes, 5=npc (trees + terrain are TILES, not objects)
    const NOBJ = 6;
    const I_PLAYER = 0; const I_E0 = 1; const NUM_EN = 4; const I_NPC = 5;
    var int ox[6]; var int oy[6];
    var int ohp[6]; var int oactive[6]; var int oblink[6]; var int ofacing[6];
    var int ocx[6]; var int ocy[6]; var int ocw[6]; var int och[6];

    // per-map enemy spawn points, captured by apply_spawns() each time a map loads -- replaces the
    // old compile-time slime_x[]/slime_y[] consts (a map only has ONE spawn layout now, its own).
    var int en_spawn_x[NUM_EN]; var int en_spawn_y[NUM_EN];
    var int en_count;   // how many of the NUM_EN enemy slots the CURRENT map actually uses

    var int on_warp;    // edge-detect for map_warp_at: only transition on the frame you STEP ONTO it

    const PLAYER_HP = 6; const ENEMY_HP = 3; const ATTACK_REACH = 10; const STRIKE_FRAME = 1;
    // The swing anim is 4 frames * ANIM_PERIOD(200ms) = 800ms; at a real, correctly-paced 60fps
    // that's 48 logic frames. attack_lock used to be a mismatched 12 (=200ms), which raced against
    // STRIKE_FRAME's own >=200ms requirement and lost every time -- attack_lock hit 0 (and
    // resolve_attack's "if attack_lock<=0 return" bailed) in the SAME frame af first reached 1, so
    // the hit-test never ran and the swing anim got cut back to idle after its first frame. Only
    // looked "fine" before because the simulator wasn't actually holding a clean 60fps.
    const ATTACK_LOCK = 48;
    const IFRAMES = 40; const HURT_BLINK = 20; const KB = 6; const RESPAWN_T = 120;
    const F_NPC_MET = 0;
    const DIR_SIDE = 0; const DIR_UP = 1; const DIR_DOWN = 2;   // matches map entry "facing" encoding

    var int player_dir; var int player_iframes; var int attack_lock; var int attack_fired; var int flash_timer; var int flash_restore;
    var int en_dir[4]; var int en_timer[4]; var int en_respawn[4];
    var int dialog_on; var int prev_b; var int dialog_dirty; var int dialog_first_time;

    void set_collider(int i, int cx, int cy, int cw, int ch) { ocx[i] = cx; ocy[i] = cy; ocw[i] = cw; och[i] = ch; }

    int obj_hit(int a, int b) {
        return aabb(ox[a] + ocx[a], oy[a] + ocy[a], ocw[a], och[a],
                    ox[b] + ocx[b], oy[b] + ocy[b], ocw[b], och[b]);
    }

    // TILE collision: lower-body footprint vs solid tiles (trees + water) in the CURRENTLY LOADED
    // map (cur_world/cur_w/cur_flags -- lib/map.lib globals, set by map_load).
    int tile_blocked(int px, int py) {
        if map_solid_at(cur_world, cur_w, cur_flags, px + 3,  py + 10) == 1 { return 1; }
        if map_solid_at(cur_world, cur_w, cur_flags, px + 12, py + 10) == 1 { return 1; }
        if map_solid_at(cur_world, cur_w, cur_flags, px + 3,  py + 15) == 1 { return 1; }
        if map_solid_at(cur_world, cur_w, cur_flags, px + 12, py + 15) == 1 { return 1; }
        return 0;
    }
    int move_obj(int i, int dx, int dy) {
        var int nx; var int ny; var int blk; var int ww; var int wh;
        ww = cur_w * 16; wh = cur_h * 16;   // world pixel bounds of the CURRENT map (dims vary per map)
        blk = 0;
        nx = ox[i] + dx; if nx < 0 { nx = 0; } if nx > ww - 16 { nx = ww - 16; }
        if tile_blocked(nx, oy[i]) == 0 { ox[i] = nx; } else { blk = 1; }
        ny = oy[i] + dy; if ny < 0 { ny = 0; } if ny > wh - 16 { ny = wh - 16; }
        if tile_blocked(ox[i], ny) == 0 { oy[i] = ny; } else { blk = 1; }
        return blk;
    }

    void respawn_enemy(int e) {
        var int oi;
        oi = I_E0 + e;
        ox[oi] = en_spawn_x[e]; oy[oi] = en_spawn_y[e];   // back to THIS map's painted spawn
        ohp[oi] = ENEMY_HP; oactive[oi] = 1; oblink[oi] = 0;
        anim_play(oi, S_E_SIDE, 5, ANIM_PERIOD);
        en_dir[e] = rand_range(4); en_timer[e] = 40 + rand_range(60);
    }

    // ---- map switching: despawn everything but the player, then repopulate from the new map's
    // spawn list (engine-agnostic: map_load only hands back {type,tx,ty,arg} tuples -- the GAME
    // decides what each type means, here SPAWN_SLIME/SPAWN_NPC from arena_maps.gen.txt) ----
    void despawn_all() {
        var int i;
        i = I_E0;
        while i < I_E0 + NUM_EN { oactive[i] = 0; oblink[i] = 0; anim_clear(i); i = i + 1; }
        oactive[I_NPC] = 0; oblink[I_NPC] = 0; anim_clear(I_NPC);
        en_count = 0;
    }
    void apply_spawns() {
        var int i; var int type; var int tx; var int ty; var int arg; var int oi;
        despawn_all();
        i = 0;
        while i < map_spawn_count() {
            type = map_spawn_field(i, 0); tx = map_spawn_field(i, 1); ty = map_spawn_field(i, 2); arg = map_spawn_field(i, 3);
            if type == SPAWN_SLIME {
                if en_count < NUM_EN {
                    oi = I_E0 + en_count;
                    en_spawn_x[en_count] = tx * 16; en_spawn_y[en_count] = ty * 16;
                    ox[oi] = en_spawn_x[en_count]; oy[oi] = en_spawn_y[en_count];
                    ohp[oi] = ENEMY_HP; oactive[oi] = 1; oblink[oi] = 0;
                    set_collider(oi, 4, 11, 8, 4); anim_play(oi, S_E_SIDE, 5, ANIM_PERIOD);
                    en_dir[en_count] = rand_range(4); en_timer[en_count] = 40 + rand_range(60); en_respawn[en_count] = 0;
                    en_count = en_count + 1;
                }
            } else { if type == SPAWN_NPC {
                ox[I_NPC] = tx * 16; oy[I_NPC] = ty * 16; oactive[I_NPC] = 1;
                set_collider(I_NPC, 0, 0, 16, 16); anim_play(I_NPC, S_NPC, 1, ANIM_PERIOD);
            } }
            i = i + 1;
        }
    }

    // ---- player placement: always via an entry point (no hardcoded spawn coords) ----
    void place_player(int px, int py) { ox[I_PLAYER] = px; oy[I_PLAYER] = py; }
    void set_player_facing(int f) {
        player_dir = f;
        if f == DIR_SIDE { ofacing[I_PLAYER] = 0; }   // default to facing right; no side-facing entry used yet
    }
    void place_player_at_entry(int e) {
        place_player(map_entry_field(e, 0) * 16, map_entry_field(e, 1) * 16);
        set_player_facing(map_entry_field(e, 2));
    }
    // The warp primitive: load the new map, repopulate its enemies/npc, place the player at the
    // given entry. (No explicit "repaint" needed -- the PPU compose pipeline redraws bg/sprites
    // fresh from the tilemap/OAM every frame regardless; only the persistent text plane would need
    // clearing, and nothing here writes to it.)
    void go_to_map(int pak_id, int entry_id) {
        map_load(pak_id);
        apply_spawns();
        place_player_at_entry(entry_id);
    }

    void player_update() {
        var int dx; var int dy; var int moving; var int input; var int locked;
        input = read_input();
        if oblink[I_PLAYER] > 0 { oblink[I_PLAYER] = oblink[I_PLAYER] - 1; }
        if player_iframes > 0 { player_iframes = player_iframes - 1; }
        locked = 0;
        if attack_lock > 0 { attack_lock = attack_lock - 1; locked = 1; }
        dx = 0; dy = 0; moving = 0;
        if locked == 0 {
            if (input & 0x01) != 0 { dx = 0 - 1; moving = 1; ofacing[I_PLAYER] = 1; player_dir = DIR_SIDE; }
            if (input & 0x04) != 0 { dx = 1;     moving = 1; ofacing[I_PLAYER] = 0; player_dir = DIR_SIDE; }
            if (input & 0x02) != 0 { dy = 0 - 1; moving = 1; player_dir = DIR_UP; }
            if (input & 0x08) != 0 { dy = 1;     moving = 1; player_dir = DIR_DOWN; }
            move_obj(I_PLAYER, dx, dy);
            if (input & 0x10) != 0 { attack_lock = ATTACK_LOCK; attack_fired = 0; anim_play(I_PLAYER, S_H_ATK, 4, ANIM_PERIOD); anim_restart(I_PLAYER); }
            else {
                if player_dir == DIR_UP { anim_play(I_PLAYER, S_H_UP, 4, ANIM_PERIOD); }
                else { if player_dir == DIR_DOWN { anim_play(I_PLAYER, S_H_DOWN, 4, ANIM_PERIOD); } else { anim_play(I_PLAYER, S_H_SIDE, 4, ANIM_PERIOD); } }
                if moving == 0 { anim_restart(I_PLAYER); }
            }
        }
    }

    void enemy_update() {
        var int e; var int oi; var int dx; var int dy; var int bl;
        e = 0;
        while e < en_count {
            oi = I_E0 + e;
            if oactive[oi] == 1 {
                if oblink[oi] > 0 { oblink[oi] = oblink[oi] - 1; }
                en_timer[e] = en_timer[e] - 1;
                if en_timer[e] <= 0 { en_dir[e] = rand_range(4); en_timer[e] = 40 + rand_range(60); }
                dx = 0; dy = 0;
                if en_dir[e] == 0      { dx = 1;     ofacing[oi] = 0; anim_play(oi, S_E_SIDE, 5, ANIM_PERIOD); }
                else { if en_dir[e] == 1 { dx = 0 - 1; ofacing[oi] = 1; anim_play(oi, S_E_SIDE, 5, ANIM_PERIOD); }
                else { if en_dir[e] == 2 { dy = 0 - 1; anim_play(oi, S_E_UP, 5, ANIM_PERIOD); }
                else                     { dy = 1;     anim_play(oi, S_E_DOWN, 5, ANIM_PERIOD); } } }
                bl = move_obj(oi, dx, dy);
                if bl == 1 { en_timer[e] = 0; }
            } else {
                en_respawn[e] = en_respawn[e] - 1;
                if en_respawn[e] <= 0 { respawn_enemy(e); }
            }
            e = e + 1;
        }
    }

    void resolve_attack() {
        var int af; var int e; var int oi; var int hx; var int hy; var int hw; var int hh; var int kdx; var int kdy;
        if attack_lock <= 0 { return; }
        af = anim_elapsed(I_PLAYER); af = af / ANIM_PERIOD;
        if af < STRIKE_FRAME { return; }
        if attack_fired == 1 { return; }
        attack_fired = 1;
        kdx = 0; kdy = 0;
        if player_dir == DIR_UP { hx = ox[I_PLAYER]; hy = oy[I_PLAYER] - ATTACK_REACH; hw = 16; hh = ATTACK_REACH; kdy = 0 - KB; }
        else { if player_dir == DIR_DOWN { hx = ox[I_PLAYER]; hy = oy[I_PLAYER] + 16; hw = 16; hh = ATTACK_REACH; kdy = KB; }
        else { if ofacing[I_PLAYER] == 1 { hx = ox[I_PLAYER] - ATTACK_REACH; hy = oy[I_PLAYER]; hw = ATTACK_REACH; hh = 16; kdx = 0 - KB; }
        else { hx = ox[I_PLAYER] + 16; hy = oy[I_PLAYER]; hw = ATTACK_REACH; hh = 16; kdx = KB; } } }
        e = 0;
        while e < en_count {
            oi = I_E0 + e;
            if oactive[oi] == 1 {
                if aabb(hx, hy, hw, hh, ox[oi] + ocx[oi], oy[oi] + ocy[oi], ocw[oi], och[oi]) == 1 {
                    ohp[oi] = ohp[oi] - 1; oblink[oi] = HURT_BLINK; move_obj(oi, kdx, kdy);
                    if ohp[oi] <= 0 { oactive[oi] = 0; en_respawn[e] = RESPAWN_T; }
                }
            }
            e = e + 1;
        }
    }

    void player_hurt_check() {
        var int e; var int oi; var int kdx; var int kdy;
        if player_iframes > 0 { return; }
        e = 0;
        while e < en_count {
            oi = I_E0 + e;
            if oactive[oi] == 1 {
                if obj_hit(I_PLAYER, oi) == 1 {
                    ohp[I_PLAYER] = ohp[I_PLAYER] - 1; player_iframes = IFRAMES; oblink[I_PLAYER] = IFRAMES;
                    kdx = KB; kdy = KB;
                    if ox[I_PLAYER] < ox[oi] { kdx = 0 - KB; }
                    if oy[I_PLAYER] < oy[oi] { kdy = 0 - KB; }
                    move_obj(I_PLAYER, kdx, kdy);
                    flash_timer = 3;
                    if ohp[I_PLAYER] <= 0 { place_player_at_entry(0); ohp[I_PLAYER] = PLAYER_HP; player_iframes = IFRAMES; }
                    e = en_count;
                }
            }
            e = e + 1;
        }
    }

    int facing_npc() {
        var int fx; var int fy;
        if oactive[I_NPC] == 0 { return 0; }   // NPC isn't on every map (e.g. not in the grove)
        fx = ox[I_PLAYER]; fy = oy[I_PLAYER];
        if player_dir == DIR_UP { fy = fy - 12; }
        else { if player_dir == DIR_DOWN { fy = fy + 12; }
        else { if ofacing[I_PLAYER] == 1 { fx = fx - 12; } else { fx = fx + 12; } } }
        return aabb(fx, fy, 16, 16, ox[I_NPC], oy[I_NPC], 16, 16);
    }

    // The dialog's whole visual footprint is rows 10-15, cols 1-18 -- tile_clear on close must match
    // this exactly (border + panel), or a stray border strip is left behind.
    const DLG_X = 1;  const DLG_Y = 10; const DLG_W = 18; const DLG_H = 6;

    // dialog_first_time is decided ONCE, when the dialog opens (see the B-press trigger in main()),
    // not re-checked here every frame -- otherwise flag_get would flip mid-read (the flag gets set
    // right after the decision) and the "first meeting" text would change to the repeat line before
    // the player finishes reading it.
    // Text must stay within [DLG_X+2, DLG_X+DLG_W-2] (a margin cell on each side) -- tile_text only
    // clips to the 20-cell SCREEN edge, not to this box, so a string that runs past the box's own
    // right edge lands directly on bare tilemap (no black behind it) and, worse, sits outside the
    // tile_clear rectangle on close, leaving a stray letter behind forever. Keep every line <=14 cols.
    void draw_dialog() {
        tile_fill(DLG_X, DLG_Y, DLG_W, DLG_H, GLYPH_SOLID, C_BLACK);
        // tile_text_bg (not tile_text) so the letters' "off" pixels also paint black -- otherwise the
        // gaps between/around strokes stay transparent and show the tilemap bleeding through them.
        if dialog_first_time == 1 { tile_text_bg(3, 12, "HASTA... YOU", C_WHITE, C_BLACK); tile_text_bg(3, 13, "FOUND ME!", C_WHITE, C_BLACK); }
        else { tile_text_bg(3, 12, "I AM A SICK", C_WHITE, C_BLACK); tile_text_bg(3, 13, "NINJA.", C_WHITE, C_BLACK); }
    }

    // Rebuild + emit the Y-sorted OAM cast from the CURRENT ox/oy/anim state. Called every frame in
    // BOTH branches -- including while a dialog is open -- because cmd_reset() zeroes the OAM shadow
    // (oam_n) each frame; skipping this during dialog would emit an empty OAM command and every
    // sprite (player, ninja, enemies) would vanish until the next normal frame rebuilds it.
    // `frozen`=1 (used during dialog, since update() calls are skipped and positions don't move) holds
    // everyone on their base/idle pattern instead of anim_pat's time-based walk cycle, which would
    // otherwise keep advancing off the real clock and make the player appear to walk in place.
    void build_cast(int frozen) {
        var int i; var int pat;
        scene_clear();
        i = 0;
        while i < NOBJ {
            if oactive[i] == 1 {
                if oblink[i] > 0 { if (oblink[i] & 2) != 0 { i = i; } else {
                    if frozen == 1 { pat = anim_base_pat(i); } else { pat = anim_pat(i); }
                    if ofacing[i] == 1 { scene_obj(pat, ox[i], oy[i], 1); } else { scene_obj(pat, ox[i], oy[i], 0); }
                } }
                else {
                    if frozen == 1 { pat = anim_base_pat(i); } else { pat = anim_pat(i); }
                    if ofacing[i] == 1 { scene_obj(pat, ox[i], oy[i], 1); } else { scene_obj(pat, ox[i], oy[i], 0); }
                }
            }
            i = i + 1;
        }
        scene_draw();
    }

    // Push the CURRENT map's tiles to the PPU: a one-time upload if it fits the 32x32 window
    // (cur_stream==0, e.g. the arena), else stream the visible window every frame (the grove, 50x50).
    void push_tiles() {
        if cur_stream == 1 { scene_stream(cur_world, cur_w, cur_h); } else { ppu_scroll(scene_cam_x, scene_cam_y); }
    }

    int main() {
        var int i; var int wi;

        sys_set_fps(60);   // this game moves per-frame, so its speed is frame-rate bound: pin 60 fps

        sys_ppu_dma(PAL_MASTER, PPU_PAL);
        sys_ppu_dma(ANIM_HERO_WALK_ID,      PPU_PAT + S_H_SIDE * 256);
        sys_ppu_dma(ANIM_HERO_ATTACK_ID,    PPU_PAT + S_H_ATK * 256);
        sys_ppu_dma(ANIM_HERO_WALK_UP_ID,   PPU_PAT + S_H_UP * 256);
        sys_ppu_dma(ANIM_HERO_WALK_DOWN_ID, PPU_PAT + S_H_DOWN * 256);
        sys_ppu_dma(ANIM_ENEMY_WALK_ID,     PPU_PAT + S_E_SIDE * 256);
        sys_ppu_dma(ANIM_ENEMY_WALK_UP_ID,  PPU_PAT + S_E_UP * 256);
        sys_ppu_dma(ANIM_ENEMY_WALK_DOWN_ID,PPU_PAT + S_E_DOWN * 256);
        sys_ppu_dma(ANIM_NINJA_ID, PPU_PAT + S_NPC * 256);
        // painted terrain tiles -> the slots BOTH maps' tile ids name (assets/maps/legend.txt, 80+).
        // Both maps share this one tileset (their .map.txt omit tileset/palette -> map_load keeps
        // whatever's already loaded), so it's DMA'd once here at boot, not re-DMA'd per map switch.
        sys_ppu_dma(ANIM_CIMEN_ID, PPU_PAT + 80 * 256);   // grass
        sys_ppu_dma(ANIM_BRIDGE_ID, PPU_PAT + 81 * 256);  // walkable, over water
        sys_ppu_dma(ANIM_ROAD_ID,  PPU_PAT + 82 * 256);   // dirt path
        sys_ppu_dma(ANIM_WATER_ID, PPU_PAT + 83 * 256);   // river
        sys_ppu_dma(ANIM_TREE1_ID, PPU_PAT + 84 * 256);
        sys_ppu_dma(ANIM_TREE2_ID, PPU_PAT + 85 * 256);
        sys_ppu_dma(ANIM_TREE3_ID, PPU_PAT + 86 * 256);
        sys_ppu_dma(ANIM_TREE4_ID, PPU_PAT + 87 * 256);
        sys_ppu_dma(ANIM_TREE5_ID, PPU_PAT + T_TREE5 * 256); // ready; not yet painted into a map
        ppu_upload_font(&font8x8);

        srand_time();

        // One-time player setup (stats/collider/anim) -- NOT map-specific, so it happens once at
        // boot, unlike position (which always comes from an entry point via go_to_map below).
        ohp[I_PLAYER] = PLAYER_HP; oactive[I_PLAYER] = 1;
        set_collider(I_PLAYER, 4, 11, 8, 4); anim_play(I_PLAYER, S_H_DOWN, 4, ANIM_PERIOD);
        player_dir = DIR_DOWN; player_iframes = 0; attack_lock = 0; flash_timer = 0; flash_restore = 0;
        dialog_on = 0; prev_b = 0; dialog_dirty = 0; dialog_first_time = 0; on_warp = 0;

        go_to_map(ASSET_MAP_ARENA, 0);   // boot into the arena at entry 0 (new-game start)

        while true {
            if dialog_on == 1 {
                // Close on A. dialog_dirty is set HERE (at close), not at open -- the else branch
                // below unconditionally drains dialog_dirty every normal frame, so setting it at open
                // would clear it again on that very same frame, before the box was ever drawn, and
                // the later real close would have nothing left to trigger the cleanup wipe.
                if (read_input() & 0x10) != 0 { dialog_on = 0; dialog_dirty = 1; }
                cmd_reset();
                scene_follow(ox[I_PLAYER], oy[I_PLAYER], cur_w, cur_h);
                push_tiles();
                build_cast(1);          // world is frozen (no update calls), but still visible
                draw_dialog();
                ppu_present();
            } else {
                player_update();
                enemy_update();
                resolve_attack();
                player_hurt_check();

                // map transitions -- edge-triggered (on_warp) so standing still on a warp tile
                // doesn't re-fire it every frame.
                wi = map_warp_at(ox[I_PLAYER] >> 4, oy[I_PLAYER] >> 4);
                if wi >= 0 {
                    if on_warp == 0 { go_to_map(map_warp_field(wi, 2), map_warp_field(wi, 3)); }
                    on_warp = 1;
                } else {
                    on_warp = 0;
                }

                if (read_input() & 0x20) != 0 { if prev_b == 0 { if facing_npc() == 1 {
                    // Decide the message ONCE, before setting the flag, so a first meeting can't
                    // read back as "already met" (flag_get would flip the instant flag_set ran).
                    if flag_get(F_NPC_MET) == 0 { dialog_first_time = 1; } else { dialog_first_time = 0; }
                    flag_set(F_NPC_MET);
                    dialog_on = 1;
                } } }
                prev_b = read_input() & 0x20;

                scene_follow(ox[I_PLAYER], oy[I_PLAYER], cur_w, cur_h);
                cmd_reset();
                push_tiles();

                // The dialog box writes text-plane cells that PERSIST in PPU RAM until something
                // overwrites them -- compose_text just keeps redrawing whatever's there every frame.
                // So on the first normal frame after a dialog closes, wipe those cells back to empty
                // (glyph 0), once, or the box stays on screen forever.
                if dialog_dirty == 1 { tile_clear(DLG_X, DLG_Y, DLG_W, DLG_H); dialog_dirty = 0; }

                // sys_ppu_dma is an IMMEDIATE write to the PPU's real palette; ppu_pal1 only QUEUES
                // bytes that apply later, when ppu_present() flushes the command buffer. So the
                // restore must happen on a frame where NO ppu_pal1 white is queued -- otherwise the
                // queued white (flushed after this immediate DMA) clobbers the restore and, since
                // flash_timer never goes above 0 again, it never gets undone (stays white forever).
                if flash_timer > 0 {
                    ppu_pal1(C_GREEN, 0xFFFF); ppu_pal1(C_RED, 0xFFFF); ppu_pal1(7, 0xFFFF);
                    flash_timer = flash_timer - 1;
                    if flash_timer == 0 { flash_restore = 1; }
                } else {
                    if flash_restore == 1 { sys_ppu_dma(PAL_MASTER, PPU_PAL); flash_restore = 0; }
                }

                build_cast(0);

                tile_text(0, 0, "HP", C_WHITE);
                tile_number(3, 0, ohp[I_PLAYER], C_WHITE);

                ppu_present();
            }
        }
        return 0;
    }
