// event.lib dialog state machine (headless: the close path needs a host button, so the demo covers
// the visual/close; here we check the open/active state + style setter compile + run).
//   starts closed (+1), dialog_open sets active (+2), dialog_style runs (+4) -> 7.  Expected: 7.
include "../lib/event.lib";
{
    var byte msg[4] = { 65, 66, 67, 0 };       // "ABC"
    int main() {
        var int s;
        s = 0;
        if dialog_active() == 0 { s = s + 1; }     // starts closed
        dialog_open(&msg);
        if dialog_active() == 1 { s = s + 2; }     // open -> active
        dialog_style(2, 3, 4);                     // setter runs without error
        if dlg_panel == 2 { s = s + 4; }           // style applied
        return s;
    }
}
