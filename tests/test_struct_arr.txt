// Piece D6: array of structs (stride = sizeof(Enemy) = 3: x@0 int, hp@2 byte).
// e[i].x = i*10, e[i].hp = i+1. Return e[3].x + e[3].hp + e[0].x + e[0].hp = 30+4+0+1 = 35.
{
    struct Enemy { int x; byte hp; }
    var Enemy e[4];

    int main() {
        var int i;
        i = 0;
        while i < 4 {
            e[i].x = i * 10;
            e[i].hp = i + 1;
            i = i + 1;
        }
        return e[3].x + e[3].hp + e[0].x + e[0].hp;
    }
}
