{
    int conditional(int x) {
        var int result = 0;
        if x > 10 {
            result = 1;
        }
        if x < 5 {
            result = 2;
        }
        return result;
    }
    int main() {
        var int a = conditional(15);
        var int b = conditional(3);
        return a + b;
    }
}
