// Piece E: load_palette (bulk 512B -> PRAM) + palette_fill (set a slot range).
// Build a palette where entry k = k, load it (entry 5 -> 5), then palette_fill(10,3,0x1234)
// so entry 11 = 0x1234 (4660). 5 + 4660 = 4665. Expected return: 4665.
include "../lib/io.lib";
{
    var byte pal[512];

    int main() {
        var int i;
        i = 0;
        while i < 256 {
            pal[i * 2]     = i;     // low byte = index value
            pal[i * 2 + 1] = 0;     // high byte = 0
            i = i + 1;
        }
        load_palette(&pal);
        palette_fill(10, 3, 0x1234);
        return peek(PRAM + 5 * 2) + peek(PRAM + 11 * 2);
    }
}
