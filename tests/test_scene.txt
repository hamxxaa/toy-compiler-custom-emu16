// scene.lib smoke test: stress the 32-elem / 11-field Obj struct-array (spawn 22, round-trip fields)
// and the dead-zone camera clamp math. No rendering (that's verified in arena). All guards pass ->
// returns nobj*100 + a*10 + b = 22*100 + 0*10 + 1 = 2201.
include "../lib/scene.lib";
{
    // minimal hand-written registry: clip 1 = 16x16x4 char, clip 2 = 16x16x1 prop
    const NUM_CLIPS = 3;
    var byte clip_w[3]      = {0, 16, 16};
    var byte clip_h[3]      = {0, 16, 16};
    var byte clip_count[3]  = {0, 4, 1};
    var int  clip_period[3] = {0, 100, 100};
    var int  clip_buf[3];

    int main() {
        var int a; var int b; var int i; var int sum;
        scene_reset();
        scene_world(480, 384, 1);
        scene_deadzone(60, 44, 84, 68);

        a = spawn_obj(1, 100, 100);          // index 0
        b = spawn_obj(2, 200, 150);          // index 1
        i = 0;
        while i < 20 {                        // 20 more props -> nobj = 22, stresses the array
            spawn_obj(2, 30 + i * 10, 50 + i * 5);
            i = i + 1;
        }

        // camera clamps to world far corner: max cam = (480-160, 384-128) = (320, 256)
        camera_follow(1000, 1000);
        if cam_x != 320 { return 80001; }
        if cam_y != 256 { return 80002; }
        // camera clamps to origin
        camera_follow(0, 0);
        if cam_x != 0 { return 80003; }
        if cam_y != 0 { return 80004; }

        // struct-array round-trip: the last prop (index 21) was spawned at x=30+19*10=220, y=50+19*5=145
        if objs[21].wx != 220 { return 80005; }
        if objs[21].wy != 145 { return 80006; }
        if objs[0].clip != 1  { return 80007; }
        if objs[1].clip != 2  { return 80008; }

        // play_obj only restarts on change; obj_locked round-trip
        play_obj_once(a, 1);
        if obj_locked(a) != 1 { return 80009; }

        return nobj * 100 + a * 10 + b;
    }
}
