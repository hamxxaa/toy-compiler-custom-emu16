// switch -> O(1) jump table (Phase 0). Exercises: dense cases, a const-valued case, both
// boundaries (min/max), an interior hole and out-of-range (both -> default), and a no-default
// switch where an unmatched value falls through unchanged.
//   classify: 1->10 2->20 3->30 6->60 else->99   (min=1, max=6; 4,5 are holes)
//   nodef:    10->1 20->2 else unchanged(7)
//   10+20+30+60+99+99+99 + 1+2+7 = 427.  Expected return: 427.
{
    const HIT6 = 6;

    int classify(int x) {
        var int r;
        r = 0;
        switch (x) {
            case 1: r = 10;
            case 2: r = 20;
            case 3: r = 30;
            case HIT6: r = 60;        // 4,5 are holes -> default; const-valued case label
            default: r = 99;
        }
        return r;
    }

    int nodef(int x) {
        var int r;
        r = 7;
        switch (x) {
            case 10: r = 1;
            case 20: r = 2;
        }
        return r;                     // unmatched -> stays 7 (no default)
    }

    int main() {
        var int total;
        var int neg;
        total = 0;
        total = total + classify(1);     // 10  min boundary
        total = total + classify(2);     // 20
        total = total + classify(3);     // 30
        total = total + classify(6);     // 60  max boundary
        total = total + classify(4);     // 99  interior hole -> default
        total = total + classify(9);     // 99  above max -> default
        neg = 0 - 5;
        total = total + classify(neg);   // 99  below min -> default
        total = total + nodef(10);       // 1
        total = total + nodef(20);       // 2
        total = total + nodef(15);       // 7  unmatched, no default -> unchanged
        return total;
    }
}
