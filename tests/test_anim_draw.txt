// Sprite lib v1 end-to-end: bump allocator + blit. Streams a packed 2x2x2 sheet (companion
// test_anim_draw.manifest, asset id 0) into the pool via anim_load, checks the allocator bookkeeping
// (address set, pool_top advanced by w*h*count=8, idempotent re-load), then clip_blit frame 0 at
// (0,0) and reads it back from VRAM. frame 0 = {1,2,3,4}: (0,0)=1 (1,0)=2 (0,1)=3 (1,1)=4 ->
// 1 + 2*10 + 3*100 + 4*1000 = 4321. Expected return: 4321.
include "../lib/anim.lib";
{
    const NUM_CLIPS = 1;
    var byte clip_w[1]      = {2};
    var byte clip_h[1]      = {2};
    var byte clip_count[1]  = {2};
    var int  clip_period[1] = {125};
    var int  clip_buf[1];

    int main() {
        var int a;
        var int b;
        var int v;
        fill(0);
        anim_scene_reset();
        a = anim_load(0);                      // stream 2*2*2 = 8 bytes into the pool
        if a == 0 { return 1; }                // overflow guard
        if clip_buf[0] != a { return 2; }      // registry address recorded
        if pool_top != 8 { return 3; }         // bump advanced by w*h*count
        b = anim_load(0);                      // idempotent: same address, no re-advance
        if b != a { return 4; }
        if pool_top != 8 { return 5; }
        clip_blit(0, 0, 0, 0, 0);              // frame 0 at (0,0), facing 0 (as drawn)
        v = peek_byte(VRAM_START + 0);
        v = v + peek_byte(VRAM_START + 1) * 10;
        v = v + peek_byte(VRAM_START + 160) * 100;
        v = v + peek_byte(VRAM_START + 161) * 1000;
        return v;
    }
}
