// Piece C2: int_to_str writes "12345" (len 5, NUL-terminated); strlen reads it back as 5.
// 5*100 + 5 = 505. Expected return: 505. (Also compiles all of text.lib + io.lib.)
include "../lib/text.lib";
{
    var byte buf[8];

    int main() {
        var int len;
        len = int_to_str(&buf, 12345);
        return len * 100 + strlen(&buf);
    }
}
