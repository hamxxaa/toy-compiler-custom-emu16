include "../lib/io.lib";

// Phase-3 smoke test for the map system (plans/swapping-worldmaps-kojima.md): proves the exact
// mechanic map_load() will lean on -- reading small "header" fields out of a byte array by constant
// index, using them to build up a running offset, and re-deriving `&buf + off` at each step to reach
// a later section, then reading THAT section's bytes via a computed (non-constant) address. This
// chain (array-elem read -> offset accumulate -> address-of-array-plus-variable -> asm byte read)
// hasn't been exercised together before, even though each piece individually has.
{
    // Layout mimics a tiny headered blob: 3 "count" bytes, then 3 variable-length sections.
    var byte buf[20] = {
        3, 5, 2,                    // counts: 3 "flags" bytes, 5 "spawn" bytes, 2 "warp" bytes
        11, 22, 33,                 // flags section   (3 bytes)
        1, 2, 3, 4, 5,              // spawns section  (5 bytes)
        77, 88,                     // warps section   (2 bytes)
        0, 0, 0, 0, 0, 0, 0          // padding (unread)
    };

    int main() {
        var int off; var int flags_p; var int spawn_p; var int warp_p;
        var int flags_count; var int spawn_count; var int warp_count;
        var int sum;

        flags_count = buf[0];       // constant-index byte-array elem read
        spawn_count = buf[1];
        warp_count  = buf[2];

        off = 3;
        flags_p = &buf + off;  off = off + flags_count;   // address-of-array + int offset
        spawn_p = &buf + off;  off = off + spawn_count;   // re-based off a value READ from the array
        warp_p  = &buf + off;  off = off + warp_count;

        // Read through the computed (non-constant) addresses via the same asm byte-load path
        // ppu_rdbyte/peek_byte use (LDR reg,[addr],byte) -- exactly what map_spawn_field/etc. do.
        sum = peek_byte(flags_p) + peek_byte(flags_p + 1) + peek_byte(flags_p + 2);   // 11+22+33
        sum = sum + peek_byte(spawn_p + 2);                                            // buf[8]=3
        sum = sum + peek_byte(warp_p) + peek_byte(warp_p + 1);                        // 77+88

        return sum;   // 11+22+33+3+77+88 = 234
    }
}
