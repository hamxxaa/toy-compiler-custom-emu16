// Combat v1 engine primitives: obj_overlap_box (attack hitbox hit + miss), hp decrement, despawn
// (active flips), blink set. Player p at (50,50), enemy en at (70,50), both feet colliders (4,11,8,4).
// A right-facing hitbox (66,50,10,16) overlaps en's collider (74,61,8,4). All-pass returns 999.
include "../lib/scene.lib";
{
    const NUM_CLIPS = 3;
    var byte clip_w[3]      = {0, 16, 16};
    var byte clip_h[3]      = {0, 16, 16};
    var byte clip_count[3]  = {0, 1, 1};
    var int  clip_period[3] = {0, 100, 100};
    var int  clip_buf[3];

    int main() {
        var int p; var int en;
        scene_reset();
        scene_world(480, 384, 0);
        scene_deadzone(60, 44, 84, 68);
        p  = spawn_obj(1, 50, 50); set_collider(p, 4, 11, 8, 4);
        en = spawn_obj(2, 70, 50); set_collider(en, 4, 11, 8, 4);
        objs[en].hp = 3;

        if obj_overlap_box(en, 66, 50, 10, 16) != 1 { return 1; }   // hitbox connects
        objs[en].hp = objs[en].hp - 1;
        if objs[en].hp != 2 { return 2; }
        objs[en].hp = objs[en].hp - 2;
        if objs[en].hp > 0 { return 3; }                            // dead
        despawn(en);
        if objs[en].active != 0 { return 4; }                       // removed from play

        objs[p].blink = 20;
        if objs[p].blink != 20 { return 5; }                        // hurt-blink set

        if obj_overlap_box(en, 0, 0, 4, 4) != 0 { return 6; }       // far box misses
        return 999;
    }
}
