{
    int fibonacci(int n) {
        var int a = 0;
        var int b = 1;
        var int i = 0;
        while i < n {
            var int temp = b;
            b = a + b;
            a = temp;
            i = i + 1;
        }
        return a;
    }
    int main() {
        return fibonacci(10);
    }
}
