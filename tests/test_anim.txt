// Sprite lib v1: time-based frame math over the clip registry. clip_frame(id,e) = (e/period)%count.
// clip 0: count=4, period=125ms. elapsed 0,125,250,375 -> frames 0,1,2,3; elapsed 40125 -> frame 1
// (40125/125=321, %4=1) -- a large elapsed the OLD (elapsed*fps)/1000 would have overflowed in 16-bit.
// clip_done guards: 499 -> not done (<500), 500 -> done (>=4*125). Packed: 0 + 1*10 + 2*100 + 3*1000
// + 1*10000 = 13210. Expected return: 13210.
include "../lib/anim.lib";
{
    const NUM_CLIPS = 1;
    var byte clip_w[1]      = {2};
    var byte clip_h[1]      = {2};
    var byte clip_count[1]  = {4};
    var int  clip_period[1] = {125};
    var int  clip_buf[1];

    int main() {
        var int v;
        if clip_done(0, 499) != 0 { return 1; }    // 499 < 500 -> false (0)
        if clip_done(0, 500) != 1 { return 2; }    // 500 >= 4*125 -> true (1)
        v = clip_frame(0, 0);
        v = v + clip_frame(0, 125) * 10;
        v = v + clip_frame(0, 250) * 100;
        v = v + clip_frame(0, 375) * 1000;
        v = v + clip_frame(0, 40125) * 10000;      // no int16 overflow with period-division
        return v;
    }
}
