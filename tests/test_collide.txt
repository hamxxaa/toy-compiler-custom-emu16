// Collision v1: move-and-test (block + revert), axis independence (slide), no false positive, and
// obj_overlap (touch). a is a mover, b is solid. Each guard returns a small code on failure; all-pass
// returns 777. Colliders default to the full 16x16 sprite.
include "../lib/scene.lib";
{
    const NUM_CLIPS = 3;
    var byte clip_w[3]      = {0, 16, 16};
    var byte clip_h[3]      = {0, 16, 16};
    var byte clip_count[3]  = {0, 1, 1};
    var int  clip_period[3] = {0, 100, 100};
    var int  clip_buf[3];

    int main() {
        var int a; var int b; var int bl;
        scene_reset();
        scene_world(480, 384, 0);
        scene_deadzone(60, 44, 84, 68);

        a = spawn_obj(1, 100, 100);          // mover
        b = spawn_obj(2, 120, 100);          // solid 4px to its right (gap: 116 vs 120)
        set_solid(b, 1);

        bl = move_obj(a, 10, 0);             // into b -> blocked, reverted
        if objs[a].wx != 100 { return 1; }
        if bl != 1 { return 2; }

        move_obj(a, 0, 30);                  // down is free
        if objs[a].wy != 130 { return 3; }

        move_obj(a, 10, 0);                  // now y-separated from b -> X move succeeds
        if objs[a].wx != 110 { return 4; }

        objs[a].wx = 122; objs[a].wy = 100;  // overlap b
        if obj_overlap(a, b) != 1 { return 5; }
        objs[a].wx = 200;                     // far away
        if obj_overlap(a, b) != 0 { return 6; }

        return 777;
    }
}
