// Regression test for the SAVE (7) / LOAD (8) / SAVE_EXISTS (9) syscalls.
// Writes 8 known bytes to slot 0, clears a second buffer, reads them back, and verifies.
// Returns the byte-sum (11+22+...+88 = 396) only if save wrote 8, exists reports 1, and load
// read all 8 bytes back unchanged. The PC handler is always registered now (no --menu needed).
include "../lib/sys.lib";
{
    var byte buf[8] = { 11, 22, 33, 44, 55, 66, 77, 88 };
    var byte out[8];

    int main() {
        var int w;
        var int r;
        var int i;
        var int sum;

        w = sys_save(&buf, 8, 0);          // write 8 bytes to slot 0
        if w != 8 { return 0; }
        if sys_save_exists(0) != 1 { return 0; }

        i = 0;
        while i < 8 { out[i] = 0; i = i + 1; }   // wipe destination

        r = sys_load(&out, 8, 0);          // read them back
        if r != 8 { return 0; }

        sum = 0;
        i = 0;
        while i < 8 {
            if out[i] == buf[i] { sum = sum + out[i]; }
            i = i + 1;
        }
        return sum;                        // 396 iff every byte round-tripped
    }
}
