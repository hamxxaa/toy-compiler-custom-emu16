// Phase 3: text-plane + palette command encoding (both hosts; compose verified via checksum).
//   tile_text(2,1,"HI",5) -> 04 02 01 02 01  (40 05 FF)(41 05 FF)   ['H'=72->40, 'I'=73->41,
//                            fg=5, bg=255=PPU_NO_BG (transparent -- tile_text always emits NO_BG)]
//   ppu_pal1(3,0x1234)    -> 05 03 01 34 12
//   clen=16, bytesum=10+300+301+79=690, return=(16<<8)+690=4786
include "../lib/ppu.lib";
{
    int main() {
        var int i;
        var int sum;
        cmd_reset();
        tile_text(2, 1, "HI", 5);
        ppu_pal1(3, 0x1234);
        sum = 0; i = 0;
        while i < ppu_clen { sum = sum + ppu_cmd[i]; i = i + 1; }
        return (ppu_clen << 8) + sum;
    }
}
